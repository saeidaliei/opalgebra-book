\chapter{Selected solutions}\label{app:solutions}
\textbf{Ch.\ 3, Wintner--Wielandt.} Proof given in the text. \textbf{Ch.\ 6, last exercise.} For $\omega\ne\omega'$ the vacua are Gaussians $\psi_\omega(x)\propto\ee^{-\omega x^2/2}$ and $|\braket{0_\omega}{0_{\omega'}}|=\big(\tfrac{2\sqrt{\omega\omega'}}{\omega+\omega'}\big)^{1/2}=\alpha^{-1/2}=(1+\beta^2)^{-1/4}$ with $\alpha=\tfrac{\omega+\omega'}{2\sqrt{\omega\omega'}}$, $\beta^2=\alpha^2-1$. For $N$ oscillators the product is non-zero in the limit iff $\sum\beta_j^2<\infty$. \textbf{Ch.\ 7, ex.\ 4.} $\int_0^\infty\ee^{-\beta\lambda}\dd\lambda=1/\beta$; the other diverges. \textbf{Ch.\ 16, ex.\ 3.} $S=H_{\rm Shannon}-\log n$ for $\tau=\Tr/n$. \textbf{Ch.\ 19, ex.\ 1.} $\Omega$ with Schmidt weights $p,1-p$: $\Delta=\rho\otimes\rho^{-1}$ with $\rho=\mathrm{diag}(p,1-p)$, eigenvalues $1,1,\tfrac p{1-p},\tfrac{1-p}p$; $\sigma_t(\sigma^+)=(\tfrac p{1-p})^{\ii t}\sigma^+$, period $2\pi/|\log\tfrac p{1-p}|$. \textbf{Ch.\ 24, ex.\ 3.} For $g=\lambda\ee^{\lambda x}\theta(-x)$: $\langle x\rangle=-1/\lambda$, $h(g)=1-\log\lambda$, hence $S=1-\log\lambda-1/\lambda\le0$, with maximum $0$ at $\lambda=1$. \textbf{Ch.\ 25, ex.\ 3.} Maximize $-\beta\langle q\rangle+h(f)$ on $q\ge0$ with Lagrange multiplier for $\int f=1$: $f\propto\ee^{-\beta q}$, $\langle q\rangle=1/\beta$. \textbf{Ch.\ 35, ex.\ 1.} $\tau(\chi(x\le c))=\int_{-\infty}^c\ee^x\dd x=\ee^c$.
