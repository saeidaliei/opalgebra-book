\chapter{Notation}\label{app:notation}
The conventions below are used throughout (they are locked in \texttt{macros.tex}); \cref{ch:clpw} gives the dictionary to the CLPW paper's conventions.

\begin{center}
\begin{longtable}{@{}p{0.38\linewidth}p{0.58\linewidth}@{}}
\textbf{Symbol} & \textbf{Meaning}\\\hline
\endhead
$\R,\C,\Z,\Nat$ & number sets\\
$\HH$ & Hilbert space\\
$\BH,\ \Bnd{\HH}$ & bounded operators on $\HH$\\
$\A$ & (\cstar-)algebra, e.g.\ of a region $\A(O)$\\
$\M,\ \N$ & von Neumann algebras\\
$\M'$ & commutant of $\M$; $\M''$ bicommutant\\
$\cZ(\M)$ & centre of $\M$\\
$\id$ & identity operator\\
$a^*$ & adjoint (written \texttt{\string\ad})\\
$\Tr,\ \tr,\ \tau$ & operator trace, normalized trace, trace on a II algebra\\
$\Omega,\ \Psi_{\rm dS}$ & cyclic separating vector, Bunch--Davies state\\
$S_{\Omega}=J\Delta^{1/2}$ & Tomita operator\\
$\Delta_\Omega,\ J$ & modular operator, modular conjugation\\
$\sigma_t=\Ad\Delta^{it}$ & modular flow\\
$K=-\log\Delta$ & modular Hamiltonian ($K=\beta_{\rm dS}H$ in dS)\\
$\Delta_{\varphi,\psi}$ & relative modular operator, $S a\xi_\psi=a^*\xi_\varphi$\\
$S_{\rm rel}(\Phi\|\Psi)$ & Araki relative entropy $=-\langle\Phi|\log\Delta_{\Psi,\Phi}|\Phi\rangle$\\
$u_t$ & Connes cocycle $\Delta_{\varphi,\psi}^{it}\Delta_\psi^{-it}$\\
$\M\rtimes_\sigma\R$ & crossed product by the modular flow\\
$x,\ p=-\ii\partial_x$ & clock (energy) variable and its conjugate, $[x,p]=\ii$\\
$X=K+x$ & generator of $\lambda$ (frame 1)\\
$\sigma_p(a)=\ee^{-\ii pK}a\ee^{\ii pK}$ & dressed operator (frame 2)\\
$\theta_s$ & dual action, $X\mapsto X+s$\\
$\tau(\hat a)=\int\dd x\,\ee^x\langle a(x)\rangle$ & trace weight on $\A_{\rm cr}$\\
$P=\chi(X\le0)$ & projection onto $q\ge0$, $\tau(P)=1$\\
$q,\ H_{\rm obs}$ & observer energy; $x=-\beta_{\rm dS}q$\\
$\beta_{\rm dS}=2\pi\ell$ & de Sitter inverse temperature\\
$G_N,\ \ell$ & Newton constant, dS radius\\
$\Sgen$ & generalized entropy $A/4G_N+S_{\rm out}$\\
$h(g)=-\int g\log g$ & differential entropy of a clock density\\
$(\A,\HH,D)$ & spectral triple\\
$\Tr_\omega$ & Dixmier trace\\
\end{longtable}
\end{center}
