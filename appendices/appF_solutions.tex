\section*{Chapter~\ref{ch:intro}: Why algebras, and how to read this book}

\subsection*{\S\ref{ch:why_algebras} Why algebras?}
\textbf{\ref{ex:why_algebras:1}.} On each site $\langle\psi_\theta|\psi_{\theta'}\rangle=\cos\theta\cos\theta'+\sin\theta\sin\theta'=\cos(\theta-\theta')$, and the inner product of product vectors is the product over $N$ sites. For a local operator supported on a finite set $F$ the expectation factorizes into single-site expectations on $F$ and is $1$ on the rest, so it depends on $|F|$ sites only. $\langle\sigma^z_j\rangle=\cos^2\theta-\sin^2\theta=\cos2\theta$.\par
\textbf{\ref{ex:why_algebras:2}.} $\rho_A=\mathrm{diag}(\cos^2\alpha,\sin^2\alpha)$ and $S=-\cos^2\alpha\log\cos^2\alpha-\sin^2\alpha\log\sin^2\alpha$, which tends to $0$ like $\alpha^2(1-\log\alpha^2)$ as $\alpha\to0$. For $N$ copies the entropy is $NS(\alpha)$, which grows linearly and diverges as $N\to\infty$ for any fixed $\alpha\ne0$: no density matrix with finite entropy exists for one half of the infinite chain.\par
\textbf{\ref{ex:why_algebras:3}.} The entropy $\frac13\log(L/\epsilon)$ diverges as $\epsilon\to0$, so there is no continuum limit of the entropy at fixed $L$. Correlation functions of fixed local operators do converge, because they are expectation values of elements of the algebra; the entropy is $\Tr\rho\log\rho$, which needs a density matrix and hence a tensor-product factorization, and that structure exists only at finite cutoff.\par
\textbf{\ref{ex:why_algebras:4}.} A gauge-invariant operator must commute with the Gauss-law constraint. A Wilson line crossing the entangling surface is gauge invariant but is not a sum of products $A\otimes B$ of operators on the two sides: the electric flux through the surface belongs to both and is central. So the algebra has a centre and the Hilbert space does not factorize.\par

\subsection*{\S\ref{ch:punchline} The punchline and how to read this book}
\textbf{\ref{ex:punchline:1}.} Only the finite-dimensional type I$_n$ factors and type II$_1$ have $\tau(\id)<\infty$. Then $\id/\tau(\id)$ is a normalized density matrix with the largest possible entropy, so a maximum-entropy state exists exactly in these cases; in I$_\infty$, II$_\infty$ and III there is none.\par
\textbf{\ref{ex:punchline:2}.} $\rho^{it}\in M_n$ is unitary (functional calculus) and $\sigma_t=\mathrm{Ad}\,\rho^{it}$, so $\sigma_t$ is inner. Then in the crossed product $v_t=\rho^{-it}\lambda(t)$ commutes with $M_n$ and satisfies $v_sv_t=v_{s+t}$, so $M_n\rtimes\R=M_n\otimes\{v_t\}''\cong M_n\otimes L^\infty(\R)$, which is type I and has an honest density matrix.\par

\section*{Chapter~\ref{ch:toolkit}: The mathematical toolkit}

\subsection*{\S\ref{ch:hilbert} Hilbert spaces and bounded operators}
\textbf{\ref{ex:hilbert:1}.} $\norm A^2=\lambda_{\max}(A^*A)=\norm{A^*A}$ since $\norm A$ is the largest singular value. The identity forces the norm because $\norm A=\sqrt{r(A^*A)}$ with $r$ the spectral radius, which is defined by the algebra alone (spectrum of $A^*A$).\par
\textbf{\ref{ex:hilbert:2}.} $S^ne_0=e_n$ and $(S^*)^ne_0=0$ for $n\ge1$. On $\psi=\sum c_ke_k$, $S^n\psi=\sum c_ke_{k+n}$ has $\norm{S^n\psi}=\norm\psi$ and $\norm{(S^*)^n\psi}^2=\sum_{k\ge n}|c_k|^2\to0$. So $(S^*)^n\to0$ strongly (not in norm: $\norm{(S^*)^ne_n}=1$), $S^n\to0$ only weakly, and neither in norm.\par
\textbf{\ref{ex:hilbert:3}.} For trace-class $A$ expand in an orthonormal basis and interchange the sums (absolute convergence): $\Tr(AB)=\sum_{ij}A_{ij}B_{ji}=\Tr(BA)$. For the shift $SS^*=\id-|e_0\rangle\langle e_0|$ and $S^*S=\id$: both traces are $\infty$ and $\infty-1=\infty$. Unitary invariance of the trace needs a unitary and a trace-class argument; $S$ is an isometry and none of the operators involved is trace class.\par
\textbf{\ref{ex:hilbert:4}.} Let $P$ project onto the even sites. $V:e_k\mapsto e_{2k}$ and $W:e_k\mapsto e_{2k+1}$ are isometries with $VV^*=P$, $WW^*=\id-P$, so $P\sim\id\sim\id-P$. A dimension function $d$ would give $d(\id)=d(P)+d(\id-P)=2d(\id)$, so $d(\id)\in\{0,\infty\}$: $B(\HH)$ carries no finite non-zero dimension function.\par
\textbf{\ref{ex:hilbert:5}.} Non-compact: the identity or multiplication by $\ee^{\ii x}$. Compact: any rank-one projection, or an integral operator with an $L^2$ kernel. If $\id$ were compact then the closed unit ball would be compact, forcing $\dim\HH<\infty$ (Riesz).\par

\subsection*{\S\ref{ch:topologies} Topologies on operators}
\textbf{\ref{ex:topologies:1}.} $\norm{(\id-P_n)\psi}^2=\sum_{k\ge n}|\psi_k|^2\to0$ but $\norm{\id-P_n}=1$. $\langle\varphi,S^n\psi\rangle=\sum_k\bar\varphi_{k+n}\psi_k\to0$ by Cauchy--Schwarz on tails, while $\norm{S^n\psi}=\norm\psi$. $(S^*)^n\to0$ strongly (by the same tail estimate) and $\norm{(S^*)^ne_n}=1$.\par
\textbf{\ref{ex:topologies:2}.} For a product state $\langle\sigma^z_j\sigma^z_k\rangle=m^2$ for $j\ne k$, so $\norm{(M_N-m)\Psi}^2=\frac1{N^2}\sum_j(1-m^2)=(1-m^2)/N$. For a local vector $A\Psi$, $(M_N-m)A\Psi=A(M_N-m)\Psi+[M_N,A]\Psi$ and $\norm{[M_N,A]}\le2\norm A|\mathrm{supp}A|/N\to0$. Local vectors are dense and $\norm{M_N}\le1$, so $M_N\to m$ strongly.\par
\textbf{\ref{ex:topologies:3}.} On the vector with the first $N$ spins up and the next $N$ down, $M_N=1$ and $M_{2N}=0$, so $\norm{M_N-M_{2N}}\ge1$ for all $N$: not Cauchy in norm.\par
\textbf{\ref{ex:topologies:4}.} Weakly: $\langle\varphi,f_n\psi\rangle=\int\bar\varphi\psi\,\ee^{\ii nx}\to0$ by Riemann--Lebesgue. Not strongly: $\norm{f_n\psi}=\norm\psi$. Not in norm.\par
\textbf{\ref{ex:topologies:5}.} $S^n\to0$ and $(S^*)^n\to0$ weakly, but $(S^*)^nS^n=\id$ for all $n$, which does not tend to $0$. Multiplication is only separately, not jointly, weakly continuous.\par

\subsection*{\S\ref{ch:spectral} Spectral theorem and unbounded operators}
\textbf{\ref{ex:spectral:1}.} The maximal domain $\{\int x^2|\psi|^2<\infty\}$ makes the multiplication operator equal to its adjoint, so it is self-adjoint, and $E_x(\Omega)$ is multiplication by $\chi_\Omega(x)$. $D(x^2)=\{\int x^4|\psi|^2<\infty\}$. For $\psi=(1+x^2)^{-1/2}$: $\int|\psi|^2=\pi<\infty$ but $\int x^2|\psi|^2=\int\frac{x^2}{1+x^2}=\infty$.\par
\textbf{\ref{ex:spectral:2}.} For $-\ii\psi'=\pm\ii\psi$ the solutions are $\ee^{\mp x}$. On the interval both lie in $L^2$: indices $(1,1)$, extensions $p_\theta$ with $\psi(L)=\ee^{\ii\theta}\psi(0)$ and spectrum $(\theta+2\pi n)/L$. On the half-line only $\ee^{-x}$ does: $(1,0)$, no extension. See \cref{ch:halfline}.\par
\textbf{\ref{ex:spectral:3}.} $\rho^{it}=\ee^{\ii t\log\rho}$ with $\log\rho$ self-adjoint, so it is unitary; $\rho^{is}\rho^{it}=\rho^{i(s+t)}$ and $\rho^{i0}=\id$; strong continuity is clear. The generator is $\frac{\dd}{\dd t}\rho^{it}|_0=\ii\log\rho$, i.e.\ the self-adjoint generator is $\log\rho=-K$ in later notation.\par
\textbf{\ref{ex:spectral:4}.} By the functional calculus $\ee^{-sA}$ has spectrum in $(0,1]$ so it is positive with norm $\le1$. $\ee^{+sA}$ is unbounded and defined only on $\{\psi:\int\ee^{2s\lambda}\dd\mu_\psi<\infty\}$, a dense domain of analytic-type vectors; it is not an element of $B(\HH)$.\par

\subsection*{\S\ref{ch:weyl} Weyl relations and Stone--von Neumann}
\textbf{\ref{ex:weyl:1}.} $\frac{\dd}{\dd a}\big(\ee^{\ii ap}x\ee^{-\ii ap}\big)=\ee^{\ii ap}\ii[p,x]\ee^{-\ii ap}=1$ since $[p,x]=-\ii$, so the result is $x+a$. Hence $U(a)V(b)U(a)^{-1}=\ee^{\ii b(x+a)}=\ee^{\ii ab}V(b)$, which is \eqref{eq:weyl}.\par
\textbf{\ref{ex:weyl:2}.} $\Tr[x,p]=0$ but $\Tr(\ii\id)=\ii\dim\HH\ne0$ if $\dim\HH<\infty$, so $\dim\HH=\infty$. Wintner--Wielandt is stronger: there is no solution even in infinite dimensions if $x,p$ are \emph{bounded}.\par
\textbf{\ref{ex:weyl:3}.} $V(n)=\ee^{\ii nx}$ is single-valued on the circle only for $n\in\Z$, so only then is the relation $U(a)V(n)=\ee^{\ii na}V(n)U(a)$ available. Put $U_\theta(a)=\ee^{\ii a\theta}U(a)$. An intertwiner $W$ between $\theta,\theta'$ must commute with all $V(n)$, hence is a multiplication $f(x)$, and $f(x-a)\ee^{\ii a\theta}=\ee^{\ii a\theta'}f(x)$ forces $f=\ee^{-\ii(\theta'-\theta)x}$, which is periodic iff $\theta'-\theta\in\Z$. So the $\theta\in[0,1)$ are inequivalent. On the line both $x$ and $p$ are continuous and Stone--von Neumann removes the family.\par
\textbf{\ref{ex:weyl:4}.} $\alpha^2-\beta^2=\frac{(\omega+\omega')^2-(\omega-\omega')^2}{4\omega\omega'}=1$. For the Gaussians $\psi_\omega\propto\ee^{-\omega x^2/2}$, $|\langle0_\omega|0_{\omega'}\rangle|=\big[\frac{4\omega\omega'}{(\omega+\omega')^2}\big]^{1/4}=\alpha^{-1/2}=(1+\beta^2)^{-1/4}$. For $N$ independent oscillators the overlap is $\prod_j(1+\beta_j^2)^{-1/4}$, which has a non-zero limit iff $\sum_j\log(1+\beta_j^2)<\infty$ iff $\sum\beta_j^2<\infty$.\par

\subsection*{\S\ref{ch:groups} Groups, unitary representations, and positive energy}
\textbf{\ref{ex:groups:1}.} $\int\dd t\,\ee^{-\ii tH}=\int\dd\mu(\lambda)\int\dd t\,\ee^{-\ii t\lambda}=2\pi\delta(H)$, a distribution that is not a bounded operator. For $U(1)$ with integer spectrum, $\int_0^{2\pi}\frac{\dd\theta}{2\pi}\ee^{-\ii\theta Q}=\delta_{Q,0}$, the projector onto $Q=0$.\par
\textbf{\ref{ex:groups:2}.} Every automorphism of $M_n$ is $\mathrm{Ad}\,u$ (Skolem--Noether: $M_n$ is central simple). The lattice translation $\sigma_j\mapsto\sigma_{j+1}$ cannot be $\mathrm{Ad}\,u$ with $u\in\A$: $u$ is a norm limit of local unitaries $u_N$, which commute (up to $\epsilon$) with $\sigma^z_j$ for $j$ far away, whereas the translation moves every $\sigma^z_j$ to a different operator.\par
\textbf{\ref{ex:groups:3}.} $U'(g)aU'(g)^*=U(g)zaz^*U(g)^*=U(g)aU(g)^*$ because $z\in\M'$ commutes with $a\in\M$. The ambiguity disappears when $\M'=\C\id$ (i.e.\ $\M=B(\HH)$): then $z$ is a phase.\par
\textbf{\ref{ex:groups:4}.} $\int_0^\infty\ee^{-\beta\lambda}\dd\lambda=1/\beta$, while $\int_{-\infty}^\infty\ee^{-\beta\lambda}\dd\lambda$ diverges at $\lambda\to-\infty$. The first is $\Tr\ee^{-\beta H}$ for $H\ge0$; the second is the same as the infinite total weight of $\ee^x\dd x$ on the whole line (II$_\infty$) against the finite weight on a half-line (II$_1$).\par

\subsection*{\S\ref{ch:halfline} Self-adjoint extensions and the half-line clock}
\textbf{\ref{ex:halfline:1}.} For $\psi\in D(p_{\min})$ and $\varphi\in H^1$, $\langle\varphi,p\psi\rangle-\langle p\varphi,\psi\rangle=-\ii[\bar\varphi\psi]_0^\infty=\ii\bar\varphi(0)\psi(0)=0$ since $\psi(0)=0$; so $p_{\min}$ is symmetric and $D(p_{\min}^*)\supseteq H^1$. Conversely $\varphi\in D(p^*_{\min})$ has a weak derivative in $L^2$, hence $\varphi\in H^1$. Deficiency: $p^*\psi=\ii\psi$ gives $\ee^{-x}\in L^2$, $p^*\psi=-\ii\psi$ gives $\ee^{x}\notin L^2$, so $(1,0)$.\par
\textbf{\ref{ex:halfline:2}.} The boundary form $-\ii[\bar\varphi\psi]_0^L$ vanishes for $\varphi,\psi$ obeying $\psi(L)=\ee^{\ii\theta}\psi(0)$, and the domain is maximal for this, so $p_\theta$ is self-adjoint. Eigenfunctions $\ee^{\ii\lambda x}$ with $\ee^{\ii\lambda L}=\ee^{\ii\theta}$: $\lambda=(\theta+2\pi n)/L$. A zero eigenvalue needs $\theta=0$ (periodic functions).\par
\textbf{\ref{ex:halfline:3}.} The boundary form is $\bar\varphi(0)\psi'(0)-\bar\varphi'(0)\psi(0)=\alpha\bar\varphi(0)\psi(0)-\alpha\bar\varphi(0)\psi(0)=0$ for real $\alpha$, so the operator is symmetric (and the maximal domain makes it self-adjoint). A bound state $\ee^{-\kappa x}$, $\kappa>0$, needs $-\kappa=\alpha$, hence $\alpha<0$ and $E=-\kappa^2=-\alpha^2$; for $\alpha\ge0$ no decaying solution satisfies the boundary condition.\par
\textbf{\ref{ex:halfline:4}.} $U(s)^*HU(s)=H+s$ is unitary equivalence of $H$ and $H+s$, so $\mathrm{spec}(H)=\mathrm{spec}(H)+s$ for every $s$. A non-empty closed subset of $\R$ invariant under all translations is $\R$.\par
\textbf{\ref{ex:halfline:5}.} The weight of the cut-off part is $\delta=\Phi(-q_0/\sigma)$ (Gaussian tail). Then $\norm{\psi-\psi_+}^2\approx\delta$, so for $10^{-3}$ one needs $\delta<10^{-6}$, i.e.\ $q_0/\sigma\gtrsim4.8$. The truncation is then invisible to $p=-\ii\partial_q$ up to errors of order $\ee^{-q_0^2/2\sigma^2}$, so the clock operator acts as if the line were complete.\par

\section*{Chapter~\ref{ch:cstar_states}: C$^*$-algebras, states, and representations}

\subsection*{\S\ref{ch:cstar} \cstar-algebras}
\textbf{\ref{ex:cstar:1}.} $\norm a^2=\norm{a^*a}\le\norm{a^*}\norm a$ gives $\norm a\le\norm{a^*}$, and by symmetry equality. For $a=a^*$: $\norm{a^2}=\norm{a^*a}=\norm a^2$, and inductively $\norm{a^{2^k}}=\norm a^{2^k}$, so $\norm a=\lim\norm{a^n}^{1/n}=r(a)$. For general $a$, $\norm a=r(a^*a)^{1/2}$: the norm is fixed by the algebraic structure.\par
\textbf{\ref{ex:cstar:2}.} $\norm{a\otimes\id}=\norm a$, so the inclusions are isometric and the inductive-limit norm is well defined. $\sigma^z_j$ and $\sigma^x_k$ act on different tensor factors and commute. $M_N\in\A_N\subset\A$ for each finite $N$, but $\norm{M_N-M_{2N}}=1$ (take all spins aligned with the sign pattern), so it is not norm Cauchy.\par
\textbf{\ref{ex:cstar:3}.} The spectrum of $a(x)=x$ is $[0,1]$ and $\norm a=1$. $\chi_{[0,1/2]}(a)$ is discontinuous, hence not in $C[0,1]$. For a closed ideal $I\subset C(X)$ put $Y=\{x:f(x)=0\ \forall f\in I\}$; then $I\subset I_Y$, and by a partition-of-unity argument (Urysohn) every $f\in I_Y$ is a limit of elements of $I$, so $I=I_Y$.\par
\textbf{\ref{ex:cstar:4}.} From $aa^*+a^*a=\norm f^2$ and $a^2=0$: $(a^*a)^2=a^*(\norm f^2-a^*a)a=\norm f^2a^*a$, so $a^*a/\norm f^2$ is a projection and $\norm a^2=\norm{a^*a}=\norm f^2$. Fermion operators are automatically bounded, so the CAR algebra is a \cstar-algebra directly, with no need for Weyl exponentials.\par

\subsection*{\S\ref{ch:states} States}
\textbf{\ref{ex:states:1}.} Positivity gives $0\le\omega(a^*a)+\lambda\omega(a^*b)+\bar\lambda\omega(b^*a)+|\lambda|^2\omega(b^*b)$ for all $\lambda$. Choosing $\lambda=-\omega(b^*a)/\omega(b^*b)$ yields $|\omega(b^*a)|^2\le\omega(a^*a)\omega(b^*b)$.\par
\textbf{\ref{ex:states:2}.} States on $M_2$ are $\rho=\frac12(\id+\vec r\cdot\vec\sigma)$ with $|\vec r|\le1$; pure iff $|\vec r|=1$. A mixed state is the midpoint of infinitely many chords through $\vec r$, hence has infinitely many decompositions into pure states. For the classical bit $\C^2$ the state space is a segment (a simplex), where every point has a unique decomposition.\par
\textbf{\ref{ex:states:3}.} $\omega_\psi$ is the vector state of a unit vector in an irreducible representation, hence pure on $M_2\otimes M_2$. Its restriction to $M_2\otimes\id$ is $\mathrm{diag}(\cos^2\alpha,\sin^2\alpha)$, mixed unless $\alpha\in\{0,\pi/2\}$, with entropy $-\cos^2\alpha\log\cos^2\alpha-\sin^2\alpha\log\sin^2\alpha$.\par
\textbf{\ref{ex:states:4}.} On each $M_{2^N}$ the restriction of a tracial state is a tracial state of a matrix algebra, which is unique and equals $\Tr/2^N$. These are consistent under the inclusions, so the state is determined on the union and, by continuity, on $\A$. It is clearly tracial.\par
\textbf{\ref{ex:states:5}.} Normal states are $\Tr(\rho\,\cdot)$, a convex set that is not weak$^*$ closed. Since $\dim\HH=\infty$ the Calkin algebra $B/\mathcal K$ is non-zero and has a state; composing with the quotient map gives a state $\omega$ vanishing on every compact operator. A normal state would have $\sum_i\omega(|e_i\rangle\langle e_i|)=1$, but here every term is $0$: $\omega$ is singular.\par

\subsection*{\S\ref{ch:gns} The GNS construction}
\textbf{\ref{ex:gns:1}.} (a) $N_\omega=\{a:a\psi=0\}$ (2-dimensional), $\HH_\omega=M_2/N_\omega\cong\C^2$, $\pi_\omega(\A)'=\C$, $\Omega=\psi$ is cyclic but not separating. (b) $\omega$ is faithful, $N_\omega=0$, $\HH_\omega=M_2$ with $\langle a,b\rangle=\Tr(\rho a^*b)$ (dimension 4), $\pi_\omega(\A)'\cong M_2$ (right multiplications), $\Omega=\id$ is cyclic and separating.\par
\textbf{\ref{ex:gns:2}.} $\langle f,g\rangle=\int\bar fg\,\dd\mu$ so $\HH_\omega=L^2(X,\mu)$ with $\pi_\omega(f)$ multiplication and $\Omega=1$. The commutant of $C(X)$ acting by multiplication on $L^2(\mu)$ is $L^\infty(X,\mu)$ (maximal abelian). $\pi_\omega$ is irreducible iff the commutant is $\C$, iff $\mu$ is a point mass.\par
\textbf{\ref{ex:gns:3}.} By definition $\HH_\tau$ is the completion of $\A/N_\tau$ in $\tau(a^*a)^{1/2}$ and $N_\tau=0$. Right multiplication $\xi\mapsto\xi b$ is bounded (trace property) and commutes with left multiplication, and equals the commutant; $J\xi=\xi^*$ implements an anti-isomorphism $\pi_\tau(\A)''\to\pi_\tau(\A)'$. The weak closure is the hyperfinite II$_1$ factor.\par
\textbf{\ref{ex:gns:4}.} $\HH_\omega=M_n$ with $\langle a,b\rangle=\Tr a^*b$. $\Omega=\rho^{1/2}$ is cyclic since $a\rho^{1/2}$ spans $M_n$ for invertible $\rho$, and separating since $a\rho^{1/2}=0\Rightarrow a=0$. For $\xi=a\rho^{1/2}$, $S\xi=a^*\rho^{1/2}=\rho^{-1/2}\xi^*\rho^{1/2}$, and with $J\xi=\xi^*$ this is $S=J\,(L_{\rho^{1/2}}R_{\rho^{-1/2}})=J(\rho^{1/2}\otimes\rho^{-1/2})$ (because $JL_aR_bJ=L_{b^*}R_{a^*}$). Hence $S^*S=\rho\otimes\rho^{-1}$.\par

\subsection*{\S\ref{ch:inequiv} Inequivalent representations}
\textbf{\ref{ex:inequiv:1}.} Choose phases so that $\langle\psi_j,\psi_j'\rangle\ge0$. Let $\Psi'_N$ equal $\Psi'$ on the first $N$ sites and $\Psi$ beyond; it is a local unitary rotation of $\Psi$. $\langle\Psi'_N,\Psi'\rangle=\prod_{j>N}\langle\psi_j,\psi_j'\rangle\to1$ because $\sum(1-|\langle\psi_j,\psi_j'\rangle|)<\infty$. So $\Psi'$ is in the closure of $\A\Psi$.\par
\textbf{\ref{ex:inequiv:2}.} In the representation of $\omega_p$ one has $M_N\to2p-1$ strongly, but in that of $\tau=\omega_{1/2}$, $M_N\to0$. The strong limit lies in the centre of $\pi(\A)''$ and distinguishes the representations, so they are disjoint. A state with a finite density of excitations over $\tau$ therefore cannot be reached by local operations.\par
\textbf{\ref{ex:inequiv:3}.} The flip $F$ gives $\omega_\theta\circ F(\sigma^z)=-\cos2\theta\ne\cos2\theta$ for $\theta<\pi/4$, so $\omega_\theta\circ F\ne\omega_\theta$. If a unitary on $\HH_\theta$ implemented $F$ then $\omega_\theta\circ F$ would be a vector state of $\pi_\theta$, hence quasi-equivalent to $\omega_\theta$, but in it $M_N\to-\cos2\theta$ whereas in $\pi_\theta$ it tends to $+\cos2\theta$: contradiction.\par
\textbf{\ref{ex:inequiv:4}.} For $\dim\HH<\infty$, $B(\HH)$ has a unique irreducible representation; every automorphism is inner, hence unitarily implemented, and there are no inequivalent ground states to choose between.\par

\subsection*{\S\ref{ch:quasilocal} Quasi-local algebras and the thermodynamic limit}
\textbf{\ref{ex:quasilocal:1}.} $[H,\sigma^\pm_j]=\pm h\sigma^\pm_j$ so $\alpha_t(\sigma^\pm_j)=\ee^{\pm\ii ht}\sigma^\pm_j$ and $\alpha_t$ exists trivially. The Gibbs state of $H_\Lambda$ is the product $\bigotimes\ee^{-\beta h\sigma^z/2}/Z_1$, so $\omega_\beta(\sigma^z)=\frac{\ee^{-\beta h/2}-\ee^{\beta h/2}}{\ee^{\beta h/2}+\ee^{-\beta h/2}}=-\tanh(\beta h/2)$. It is invariant: $\omega_\beta(\sigma^\pm)=0$ and the remaining monomials are diagonal.\par
\textbf{\ref{ex:quasilocal:2}.} $\norm{H_N-H_M}=\frac h2|N-M|$ (the operator $\sum_{M<j\le N}\sigma^z_j$ has norm $N-M$), so $H_N$ is not Cauchy and $H\notin\A$. In the representation $\pi_\beta$ the generator is the self-adjoint Liouvillean; it is unbounded below because one can flip arbitrarily many spins and gain energy $h$ each time (the spectrum is all of $\R$ at $\beta<\infty$).\par
\textbf{\ref{ex:quasilocal:3}.} The all-up and all-down product states each minimize every $H_\Lambda$. $M_N\to+1$ in one and $-1$ in the other, so they are disjoint. The derivation $\delta(a)=\ii[H_\Lambda,a]$ for local $a$ is defined by the same finite sums in both, hence is the same.\par
\textbf{\ref{ex:quasilocal:4}.} For disjoint states $\norm{\omega-\omega'}=2$. For pure product states with different $\theta$, on $N$ sites $\norm{\omega_N-\omega'_N}=2\sqrt{1-\cos^{2N}(\theta-\theta')}\to2$. So the sequence $\omega_\Lambda$ cannot converge in norm and only weak$^*$ convergence (on local observables) is available.\par

\section*{Chapter~\ref{ch:vn}: von Neumann algebras}

\subsection*{\S\ref{ch:doublecomm} Weak closure and the double commutant theorem}
\textbf{\ref{ex:doublecomm:1}.} For $\M=M_n\otimes\id_k$ the commutant is $\id_n\otimes M_k$ and $\M''=\M$; the centre is $\C$. For $\{\mathrm{diag}(a,a,b)\}\subset M_6$, $\M\cong(M_2\otimes\id_2)\oplus M_2$ and $\M'=(\id_2\otimes M_2)\oplus\C\id_2$ (dimension $4+1$), with $\M''=\M$ and centre $\M\cap\M'=\{\lambda\id_4\oplus\mu\id_2\}$ of dimension 2.\par
\textbf{\ref{ex:doublecomm:2}.} A finite-dimensional $^*$-algebra is $\bigoplus_kM_{n_k}$, and any representation is $\bigoplus_k\C^{n_k}\otimes\C^{m_k}$ with $M_{n_k}$ acting on the first factor. The commutant is then $\bigoplus_k\id_{n_k}\otimes M_{m_k}$.\par
\textbf{\ref{ex:doublecomm:3}.} Take continuous $f_n$ with $0\le f_n\le1$, $f_n=1$ on $[0,\frac12-\frac1n]$ and $0$ on $[\frac12,1]$. Dominated convergence gives $\norm{(f_n-\chi)\psi}^2=\int|f_n-\chi|^2|\psi|^2\to0$, so $f_n\to\chi$ strongly, hence $\chi\in\A''$. The convergence is not in norm since $\norm{f_n-\chi}_\infty=1$.\par
\textbf{\ref{ex:doublecomm:4}.} If $\M\subset\mathcal N$ then $\mathcal N'\subset\M'$. Also $\M\subset\M''$ gives $\M'''\subset\M'$, and applying the first statement to $\M'\subset(\M')''$ gives $\M'\subset\M'''$. So $\M'''=\M'$. The product statement is Tomita's commutation theorem.\par

\subsection*{\S\ref{ch:factors} Projections, factors, and the centre}
\textbf{\ref{ex:factors:1}.} $\cZ(\M)=\M\cap\M'$ and $\cZ(\M')=\M'\cap\M''=\M'\cap\M$. For $\bigoplus_iM_{n_i}\otimes\id_{m_i}$ the centre is spanned by the block identities, so the central projections are sums of block identities.\par
\textbf{\ref{ex:factors:2}.} $\M=\bigoplus_qB(\HH_q)$: its centre is spanned by the sector projections $P_q$, so it is not a factor. Charged operators map $\HH_q\to\HH_{q'}$ and live outside $\M$ (they are in $B(\HH)$ but not in $\M$). The commutant is $\M'=\bigoplus_q\C\id_q=L^\infty(\Z)$, an abelian algebra generated by functions of $Q$.\par
\textbf{\ref{ex:factors:3}.} ($\Leftarrow$) If $\M'\xi$ is dense and $a\xi=0$ with $a\in\M$, then $aM'\xi=M'a\xi=0$, so $a=0$ on a dense set. ($\Rightarrow$) Let $P$ be the projection onto $\overline{\M'\xi}$; it commutes with $\M'$, so $P\in\M''=\M$. Since $\xi\in\overline{\M'\xi}$, $(\id-P)\xi=0$ with $\id-P\in\M$, and separating gives $P=\id$.\par
\textbf{\ref{ex:factors:4}.} The product vector state is pure on the UHF algebra, so its GNS representation is irreducible and $\pi_\theta(\A)''=\Bnd{\HH_\theta}$, a factor of type I$_\infty$. For the mixture $\frac12(\omega_\theta+\omega_{\theta'})$ the representations are disjoint, the GNS space is $\HH_\theta\oplus\HH_{\theta'}$, the weak closure is $\Bnd{\HH_\theta}\oplus\Bnd{\HH_{\theta'}}$ and its centre is $\C^2$, spanned by the two summand projections.\par

\subsection*{\S\ref{ch:traces} Normal states, weights, and traces}
\textbf{\ref{ex:traces:1}.} If $e$ is the projection onto $\ker\rho$ then $\omega(e)=\tau(\rho e)=0$, so faithfulness gives $e=0$ and $\rho$ has trivial kernel (it may fail to be boundedly invertible in infinite dimensions). For $M_n$ with $\Tr$, $S(\omega)=-\Tr\rho\log\rho$; for a product state on $M_n\otimes M_k$, $\rho=\rho_1\otimes\rho_2$ and $S=S(\rho_1)+S(\rho_2)$.\par
\textbf{\ref{ex:traces:2}.} $\phi(u)=u\log u$ is convex and $\tau$ is a probability measure on the spectrum, so $\tau(\phi(\rho))\ge\phi(\tau(\rho))=\phi(1)=0$, i.e.\ $S(\omega)=-\tau(\rho\log\rho)\le0$, with $S(\tau)=0$ for $\rho=\id$.\par
\textbf{\ref{ex:traces:3}.} With $\mathrm{spec}H=\R$ and $\Tr f(H)=\int f\,\dd\lambda$, $\varphi(\id)=\int_\R\ee^{-\beta\lambda}\dd\lambda=\infty$, so it is not a state, but $\varphi(P_{[a,b]})=\int_a^b\ee^{-\beta\lambda}\dd\lambda<\infty$ on spectral projections (semifinite), it is normal, and faithful because $\ee^{-\beta H}>0$. Its modular flow is $\mathrm{Ad}\,\ee^{-\ii\beta Ht}$, which is not the identity since $H$ is not scalar, so $\varphi$ is not a trace.\par
\textbf{\ref{ex:traces:4}.} If $\tau_2=\tau_1(h\,\cdot)$ with $h\ge0$ affiliated to $\M$, the trace property of both forces $h$ to commute with every element, so $h\in\cZ(\M)=\C$ and $\tau_2=c\,\tau_1$.\par

\subsection*{\S\ref{ch:classification} Classification: types I, II, III}
\textbf{\ref{ex:classification:1}.} Equal rank means the two ranges have orthonormal bases of the same cardinality, and a bijection of bases defines a partial isometry with $v^*v=e$, $vv^*=f$; conversely a partial isometry preserves rank. The even-site projection $P$ is equivalent to $\id$ by $e_k\mapsto e_{2k}$ but $P\ne\id$. It is not finite: a projection is finite iff it is not equivalent to a proper subprojection of itself, and $P$ is.\par
\textbf{\ref{ex:classification:2}.} A rank-$k$ projection in $M_{2^n}$ has $\tau=k/2^n$. The dyadic rationals are dense and \cref{ch:dimension} builds, for any real $r$, an increasing sequence of projections with $\tau\to r$, whose strong limit has trace $r$. A minimal projection would have a smallest positive trace, but every projection of trace $t$ splits into two of trace $t/2$.\par
\textbf{\ref{ex:classification:3}.} The trace of a factor $M_n$ is unique (Skolem--Noether/Schur). With $\rho=n\,\mathrm{diag}(p_i)$, $\omega(a)=\Tr(\mathrm{diag}(p)\,a)$ and $S=-\tau(\rho\log\rho)=-\sum_ip_i\log(np_i)=H_{\rm Shannon}-\log n$.\par
\textbf{\ref{ex:classification:4}.} In type III every non-zero projection is equivalent to $\id$, so there are no minimal projections; a pure normal state would have a minimal support projection, so none exists. A semifinite trace would give $\tau(\id)=\tau(e)+\tau(f)=2\tau(\id)$ for orthogonal $e,f\sim\id$, so $\tau(\id)\in\{0,\infty\}$ and the trace cannot be finite on any non-zero projection.\par
\textbf{\ref{ex:classification:5}.} Suppose $\A(O)$ were type I, $B(K)\otimes\id$. The vacuum restricted to it would be a normal state with a density matrix $\rho$, and its modular operator would be $\Delta=\rho\otimes\rho'^{-1}$, which has pure point spectrum on $K\otimes K'$. But for a wedge, Bisognano--Wichmann gives $\Delta^{it}=\ee^{-2\pi\ii tB}$ with $B$ the boost generator, whose spectrum is absolutely continuous; so wedge algebras are not type I (and double cones in a CFT are conformal images of wedges). For general regions see \cref{ch:typeIII_local}.\par

\subsection*{\S\ref{ch:subsystems} Subsystems and entanglement}
\textbf{\ref{ex:subsystems:1}.} $E$ is positive, unital and an $\N$-bimodule map (it is the identity on $a\otimes\id$), so it is a conditional expectation onto $\N$. $\omega_{AB}\circ E=\omega_{AB}$ means $\omega_{AB}(a\otimes b)=\omega_{AB}(a\,\omega_B(b)\otimes\id)=\omega_A(a)\omega_B(b)$, i.e.\ $\omega_{AB}$ is the product state; the converse is immediate.\par
\textbf{\ref{ex:subsystems:2}.} $\sigma_t(a)=\rho^{it}a\rho^{-it}$, so $\sigma_t(\N)=\N$ iff $\N$ is invariant under conjugation by $\rho^{it}$. For $\N=M_k\otimes\id$ and $\rho=\rho_A\otimes\rho_B$, $\rho^{it}=\rho_A^{it}\otimes\rho_B^{it}$ maps $a\otimes\id\mapsto\rho_A^{it}a\rho_A^{-it}\otimes\id\in\N$. The $\omega$-preserving expectation is $E(X)=\Tr_B[(\id\otimes\rho_B)X]\otimes\id$, the partial trace weighted by $\rho_B$ (the Heisenberg-picture counterpart of the usual partial trace on states).\par
\textbf{\ref{ex:subsystems:3}.} Take $A_0=\sigma_z,A_1=\sigma_x$, $B_{0,1}=(\sigma_z\pm\sigma_x)/\sqrt2$. For $(\ket{\uparrow\uparrow}+\ket{\downarrow\downarrow})/\sqrt2$ one has $\langle\sigma_z\sigma_z\rangle=\langle\sigma_x\sigma_x\rangle=1$ and $\langle\sigma_z\sigma_x\rangle=0$, so each of the four terms contributes $1/\sqrt2$ (with the sign in front of $A_1B_1$ absorbed) and the sum is $4/\sqrt2=2\sqrt2$. For type III algebras Summers--Werner type theorems (under their hypotheses) give maximal violation in \emph{every} normal state, not just a specially prepared one.\par
\textbf{\ref{ex:subsystems:4}.} $S\sim c\,r^2/\delta^2$ is the area $4\pi r^2$ over the cutoff area $\delta^2$. The split property requires a finite collar $\delta>0$ between the algebras, and the type I interpolating factor has entropy $\sim r^2/\delta^2$ that diverges as $\delta\to0$: the limit does not exist, reflecting that the sharp-region algebra is type III and has no entropy.\par

\subsection*{\S\ref{ch:dimension} Dimension and the hyperfinite II$_1$ factor}
\textbf{\ref{ex:dimension:1}.} If $e=v^*v$, $f=vv^*$ then $v$ maps $\mathrm{ran}\,e$ isometrically onto $\mathrm{ran}\,f$, so ranks agree; conversely equal ranks give such a $v$ (map orthonormal bases). With $\tau=\mathrm{rank}/n$ this is the same as $\tau(e)=\tau(f)$.\par
\textbf{\ref{ex:dimension:2}.} Write $\frac13=\sum_{k\ge1}4^{-k}$ and take $t_{2m}=\sum_{k\le m}4^{-k}$: $t_2=\frac14$, $t_4=\frac5{16}$, $t_6=\frac{21}{64}$. Stage 1: a rank-1 projection $e_1\in M_4$. Stage 2: add a rank-4 projection orthogonal to $e_1\otimes\id_4$ in $M_{16}$ (total rank 5). Stage 3: add a rank-16 projection orthogonal to the previous one in $M_{64}$ (total rank 21). The strong limit has $\tau=\frac13$.\par
\textbf{\ref{ex:dimension:3}.} The reduced density matrix of a vector in $\C^n\otimes\C^m$ on $M_n$ has rank at most $m$, so $S_{\rm vN}\le\log m$ and $S_\tau=S_{\rm vN}-\log n\le\log(m/n)$. Equality holds iff the reduced state is $P/m$ for a rank-$m$ projection $P$, i.e.\ for maximally entangled states between an $m$-dimensional subspace of $\C^n$ and $\C^m$.\par
\textbf{\ref{ex:dimension:4}.} If $\omega$ were pure and normal, $\omega$ would be faithful on $s\M s$ with $s$ its support, and purity forces $s\M s=\C s$, i.e.\ $s$ minimal. But in a II$_1$ factor every non-zero projection $p$ contains a subprojection of trace $\tau(p)/2$, so there are no minimal projections.\par
\textbf{\ref{ex:dimension:5}.} With $\tau(\id)=1$, $\tau(\id-e)=1-\tau(e)=1-\tau(f)=\tau(\id-f)$, and the criterion gives $\id-e\sim\id-f$. In a II$_\infty$ factor $\tau(\id)=\infty$ and $\tau(\id-e)=\infty-\tau(e)$ is undefined for infinite $e$: take $e=\id$ and $f=\id\otimes p$ (infinite but proper). Then $e\sim f$ (both infinite) but $\id-e=0\not\sim\id-f$.\par

\section*{Chapter~\ref{ch:modular}: Equilibrium and modular theory}

\subsection*{\S\ref{ch:kms} KMS equilibrium}
\textbf{\ref{ex:kms:1}.} With $\rho=\ee^{-\beta H}/Z$, $F(z)=\Tr(\rho\,a\,\ee^{\ii zH}b\,\ee^{-\ii zH})$ is entire in finite dimensions, and $F(t+\ii\beta)=\frac1Z\Tr(\ee^{-\beta H}a\,\ee^{-\beta H}\alpha_t(b)\ee^{\beta H})=\frac1Z\Tr(\alpha_t(b)\,a\,\ee^{-\beta H})=\omega(\alpha_t(b)a)$ by cyclicity. Stationarity follows because $\rho$ commutes with $H$, so $\omega(\alpha_t(a))=\Tr(\rho\,\ee^{\ii tH}a\ee^{-\ii tH})=\omega(a)$.\par
\textbf{\ref{ex:kms:2}.} $[H,\sigma^+]=h\sigma^+$ gives $\alpha_t(\sigma^+)=\ee^{\ii ht}\sigma^+$. Hence $\omega(\sigma^-\alpha_t(\sigma^+))=\ee^{\ii ht}\omega(\sigma^-\sigma^+)=:F(t)$, entire in $t$, and $\omega(\alpha_t(\sigma^+)\sigma^-)=\ee^{\ii ht}\omega(\sigma^+\sigma^-)$. The strip condition $F(t+\ii\beta)=\ee^{\ii ht}\ee^{-\beta h}\omega(\sigma^-\sigma^+)$ must equal $\ee^{\ii ht}\omega(\sigma^+\sigma^-)$, so $\omega(\sigma^+\sigma^-)/\omega(\sigma^-\sigma^+)=\ee^{-\beta h}$; equivalently $\omega(\sigma^-\sigma^+)/\omega(\sigma^+\sigma^-)=\ee^{\beta h}$, the ratio of lower to upper level populations (detailed balance).\par
\textbf{\ref{ex:kms:3}.} The modular group $\sigma_t^\omega$ is the unique flow for which $\omega$ is KMS at $\beta=-1$, so the $\beta$-KMS flow $\alpha_t$ equals $\sigma^\omega_{-t/\beta}=\mathrm{Ad}\,\Delta^{-it/\beta}$ on $\pi_\omega(\A)$. In finite dimensions $\Delta=\rho\otimes\bar\rho^{-1}=\ee^{-\beta(H\otimes\id-\id\otimes\bar H)}$ on $\HH\otimes\overline\HH$, so $\Delta^{-it/\beta}=\ee^{\ii t(H\otimes\id-\id\otimes\bar H)}$ implements $\alpha_t$.\par
\textbf{\ref{ex:kms:4}.} At $\beta=0$ the KMS condition $\omega(a\alpha_{t+\ii\beta}(b))=\omega(\alpha_t(b)a)$ reduces to $\omega(a\alpha_t(b))=\omega(\alpha_t(b)a)$. The infinite-temperature state is tracial, so this holds for every dynamics.\par

\subsection*{\S\ref{ch:tomita} Tomita--Takesaki theory}
\textbf{\ref{ex:tomita:1}.} With $\rho=\mathrm{diag}(p,1-p)$: $\Delta=\rho\otimes\rho^{-1}$ has eigenvalue $1$ on $\ket{\uparrow\uparrow},\ket{\downarrow\downarrow}$, $\frac p{1-p}$ on $\ket{\uparrow\downarrow}$, $\frac{1-p}p$ on $\ket{\downarrow\uparrow}$. $J$ swaps the factors and complex conjugates; $S=J\Delta^{1/2}$. Then $\Delta^{it}\Omega=\Omega$ and $J\Omega=\Omega$ (real coefficients), and $J\M J=\M'$. $\sigma_t(\sigma^+\otimes\id)=\big(\frac p{1-p}\big)^{\ii t}\sigma^+\otimes\id$, with period $2\pi/|\log\frac p{1-p}|$ for $p\ne\frac12$.\par
\textbf{\ref{ex:tomita:2}.} $\sigma_{-\ii}(b)=\Delta b\Delta^{-1}=\rho b\rho^{-1}$ on $\M$, so $\omega(a\sigma_{-\ii}(b))=\Tr(\rho\,a\rho b\rho^{-1})=\Tr(a\rho b)=\Tr(\rho\,ba)=\omega(ba)$. The value is $-1$ because $\sigma_t=\mathrm{Ad}\,\Delta^{\ii t}$ runs the Heisenberg flow $\mathrm{Ad}\,\ee^{\ii Ht}$ backwards: for a Gibbs state $\Delta=\ee^{-\beta(H-H')}$ gives $\sigma_t=\alpha_{-\beta t}$.\par
\textbf{\ref{ex:tomita:3}.} $\|a\Omega\|^2=\omega(a^*a)=\omega(aa^*)=\|a^*\Omega\|^2$, so $S$ is isometric on a dense set. With $S=J\Delta^{1/2}$, $\|\Delta^{1/2}\xi\|=\|\xi\|$ for all $\xi$, which for the positive operator $\Delta^{1/2}$ forces $\Delta=\id$.\par
\textbf{\ref{ex:tomita:4}.} The reduced state is $\rho_R=\ee^{-\beta H_R}/Z$, so $\Delta=\rho_R\otimes\rho_L^{-1}=\ee^{-\beta(H_R-H_L)}$ and $\Delta^{it}=\ee^{-\ii\beta t(H_R-H_L)}$. For $a\in\M_R$, $\sigma_t(a)=\ee^{-\ii\beta tH}a\,\ee^{\ii\beta tH}$. Compared with the Heisenberg flow $\ee^{\ii sH}a\,\ee^{-\ii sH}$ this is $s=-\beta t$: modular time is physical time divided by $\beta$ and runs in the opposite direction.\par
\textbf{\ref{ex:tomita:5}.} $\|\Delta^{1/2}a\Omega\|=\|J\Delta^{1/2}a\Omega\|=\|Sa\Omega\|=\|a^*\Omega\|$, so $\|\Delta^{1/2}a\Omega\|^2=\omega(aa^*)$. Also $\|\Delta^{1/2}a\Omega\|^2=\langle a\Omega,\Delta a\Omega\rangle=\langle\Omega,a^*\Delta a\Delta^{-1}\Omega\rangle=\omega(a^*\sigma_{-\ii}(a))$ since $\Delta\Omega=\Omega$. Hence $\omega(a^*\sigma_{-\ii}(a))=\omega(aa^*)$, which is the KMS condition $\omega(x\sigma_{-\ii}(y))=\omega(yx)$ with $x=a^*$, $y=a$.\par

\subsection*{\S\ref{ch:modham} Modular Hamiltonian, relative entropy, and the Connes cocycle}
\textbf{\ref{ex:modham:1}.} For $\xi_\psi=\rho_\psi^{1/2}$, $S(a\rho_\psi^{1/2})=a^*\rho_\varphi^{1/2}=J\big(\rho_\varphi^{1/2}(a\rho_\psi^{1/2})\rho_\psi^{-1/2}\big)$ with $J\xi=\xi^*$, so $S=J\,L_{\rho_\varphi^{1/2}}R_{\rho_\psi^{-1/2}}$ and $\Delta=S^*S=L_{\rho_\varphi}R_{\rho_\psi^{-1}}$, i.e.\ $\Delta X=\rho_\varphi X\rho_\psi^{-1}$. Then $\log\Delta=L_{\log\rho_\varphi}-R_{\log\rho_\psi}$ and $-\langle\xi_\psi,\log\Delta\,\xi_\psi\rangle=\Tr\rho_\psi\log\rho_\psi-\Tr\rho_\psi\log\rho_\varphi$.\par
\textbf{\ref{ex:modham:2}.} $u_t=\Delta_{\varphi,\psi}^{\ii t}\Delta_\psi^{-\ii t}$ acts as $X\mapsto\rho_\varphi^{it}\rho_\psi^{-it}X\rho_\psi^{it}\rho_\psi^{-it}$, so $u_t=\rho_\varphi^{it}\rho_\psi^{-it}\in\M$. Cocycle: $u_s\sigma_s^\psi(u_t)=\rho_\varphi^{is}\rho_\psi^{-is}\rho_\psi^{is}\rho_\varphi^{it}\rho_\psi^{-it}\rho_\psi^{-is}=\rho_\varphi^{i(s+t)}\rho_\psi^{-i(s+t)}=u_{s+t}$. Intertwining: $u_t\sigma_t^\psi(a)u_t^*=\rho_\varphi^{it}a\rho_\varphi^{-it}=\sigma_t^\varphi(a)$. If $\varphi=\psi$, $u_t=\id$; if $\rho_\varphi=1/n$, $u_t=n^{-it}\rho_\psi^{-it}$.\par
\textbf{\ref{ex:modham:3}.} Monotonicity under $\Tr_B$ is proved in \cref{ch:monotonic} from joint convexity. Equality: if $\rho=R_\sigma(\rho_A)$ with $R_\sigma(X)=\sigma^{1/2}(\sigma_A^{-1/2}X\sigma_A^{-1/2}\otimes\id)\sigma^{1/2}$ then, because $R_\sigma(\sigma_A)=\sigma$ and $R_\sigma$ is a channel, $S(\rho\|\sigma)=S(R_\sigma\rho_A\|R_\sigma\sigma_A)\le S(\rho_A\|\sigma_A)\le S(\rho\|\sigma)$, so equality holds. The converse is Petz's theorem.\par
\textbf{\ref{ex:modham:4}.} For $\xi_\psi=u\Omega$, $S_{\Omega,\psi}=uS_\Omega$ (as $S_{\Omega,\psi}(au\Omega)=a^*\Omega=u\,(au)^*\Omega$), so $\Delta_{\Omega,\psi}=S_\Omega^*u^*uS_\Omega=\Delta_\Omega$. Therefore $S(\psi\|\Omega)=-\langle\psi|\log\Delta_{\Omega,\psi}|\psi\rangle=\langle\psi|K|\psi\rangle=2\pi\langle\psi|B|\psi\rangle$. For $\psi=W(f)\Omega$ with $f$ in the right wedge the left-wedge part of $B$ vanishes and $B=\int x\,T_{00}\,\dd x\,\dd^{d-2}x_\perp$ over $x>0$ has expectation the classical energy density of the coherent profile weighted by $x$.\par

\subsection*{\S\ref{ch:connes_class} Connes classification and where type III comes from}
\textbf{\ref{ex:connes_class:1}.} (a) Fixed $p$: $\Delta$ has spectrum $\{\lambda^n\}_{n\in\Z}\cup\{0\}$, $\lambda=p/(1-p)$: type III$_\lambda$. (b) Alternating $p,q$: the ratio set is generated by $\lambda_p=\frac p{1-p}$ and $\lambda_q$; if $\log\lambda_p/\log\lambda_q$ is irrational it is dense and the type is III$_1$, if rational it is III$_\mu$ for the generator $\mu$. (c) $\sum(p_j-\frac12)^2<\infty$: by Kakutani the state is equivalent to the trace, so the factor is II$_1$.\par
\textbf{\ref{ex:connes_class:2}.} The eigenvalues of $\Delta$ on the GNS space are products of $\lambda^{\pm1}$ and $1$, i.e.\ $\lambda^n$, so $\Delta^{\ii t_0}=\ee^{\ii nt_0\log\lambda}=\ee^{\pm2\pi\ii n}=\id$ for $t_0=2\pi/|\log\lambda|$. For another faithful state $\varphi'$, Connes' cocycle gives $\sigma^{\varphi'}_t=\mathrm{Ad}\,u_t\circ\sigma_t$, so $\sigma^{\varphi'}_{t_0}=\mathrm{Ad}\,u_{t_0}$ is inner too: the period $t_0\in T(\M)$ does not depend on the state.\par
\textbf{\ref{ex:connes_class:3}.} $B=-\ii\,\dd/\dd\theta$ has plane waves $\ee^{\ii k\theta}$ as generalized eigenvectors with every real $k$, so $\mathrm{spec}B=\R$ and $\Delta=\ee^{-2\pi B}$ has spectrum $(0,\infty)$ continuous. A continuous spectrum fills the half-line (type III$_1$), unlike the discrete $\{\lambda^n\}$ of III$_\lambda$.\par
\textbf{\ref{ex:connes_class:4}.} $\theta_s(\id)=\id$, so $\tau(\id)=\tau(\theta_s(\id))=\ee^{-s}\tau(\id)$ for every $s$, forcing $\tau(\id)\in\{0,\infty\}$; for $\N\ne0$ and $\tau\ne0$ it is $\infty$.\par

\subsection*{\S\ref{ch:monotonic} Monotonicity of relative entropy}
\textbf{\ref{ex:monotonic:1}.} With $U_k\in\{\id,\sigma_x,\sigma_y,\sigma_z\}$: $\sum_kU_k\id U_k=4\cdot\id$, and for $X=\sigma_z$: $\sigma_z+\sigma_z\sigma_z\sigma_z+\ldots=\sigma_z-\sigma_z-\sigma_z+\sigma_z=0$ (similarly for $\sigma_x,\sigma_y$). So $\frac14\sum_kU_kXU_k^*=\frac12\Tr(X)\id$ on a basis, hence for all $X$.\par
\textbf{\ref{ex:monotonic:2}.} $S(\rho_a\otimes\frac\id b\|\sigma_a\otimes\frac\id b)=\Tr\rho_a(\log\rho_a-\log\sigma_a)+\Tr(\rho_a)\Tr\frac\id b\big(\log\frac1b-\log\frac1b\big)=S(\rho_a\|\sigma_a)$.\par
\textbf{\ref{ex:monotonic:3}.} $S(p\|q)=(\frac12+\delta)\ln(1+2\delta)+(\frac12-\delta)\ln(1-2\delta)=2\delta^2+\frac43\delta^4+\dots$. Pinsker's bound is $\frac12\|p-q\|_1^2=\frac12(2\delta)^2=2\delta^2$, so it is saturated to leading order.\par
\textbf{\ref{ex:monotonic:4}.} Restriction of both states from $\mathcal N_2$ to $\mathcal N_1\subset\mathcal N_2$ cannot increase relative entropy, which is \eqref{eq:monot}. Strict example: in $M_2\otimes M_2$ let $\psi=\tau$ (the trace), $\varphi$ a Bell state, $\mathcal N_1=M_2\otimes\id$. Then $S(\varphi\|\psi;\mathcal N_2)=\log4-S(\varphi)=\log4$ while on $\mathcal N_1$ both states are $\frac12\id$ and $S=0$.\par
\textbf{\ref{ex:monotonic:5}.} $S(Wp\|Wq)=\sum_j\big(\sum_iW_{ji}p_i\big)\log\frac{\sum_iW_{ji}p_i}{\sum_iW_{ji}q_i}\le\sum_{ij}W_{ji}p_i\log\frac{p_i}{q_i}=S(p\|q)$ by the log-sum inequality and $\sum_jW_{ji}=1$.\par

\section*{Chapter~\ref{ch:crossed_products}: Crossed products}

\subsection*{\S\ref{ch:crossed} The crossed product construction}
\textbf{\ref{ex:crossed:1}.} With $\pi_\alpha(a)\xi(s)=\alpha_{-s}(a)\xi(s)$ and $\lambda(t)\xi(s)=\xi(s-t)$: $[\lambda(t)\pi_\alpha(a)\lambda(t)^*\xi](s)=\alpha_{-(s-t)}(a)\xi(s)=\alpha_{-s}(\alpha_t(a))\xi(s)=[\pi_\alpha(\alpha_t(a))\xi](s)$. $\pi_\alpha(ab)=\pi_\alpha(a)\pi_\alpha(b)$ since each $\alpha_{-s}$ is an automorphism, and $\pi_\alpha(a^*)=\pi_\alpha(a)^*$ since each $\alpha_{-s}$ preserves adjoints.\par
\textbf{\ref{ex:crossed:2}.} $\frac{\dd}{\dd c}(\ee^{\ii cp}x\ee^{-\ii cp})=\ee^{\ii cp}\ii[p,x]\ee^{-\ii cp}=1$ gives $x+c$. With $c=-K$: $\ee^{-\ii pK}(K+x)\ee^{\ii pK}=K+(x-K)=x$, since $K$ commutes with $p$-conjugation up to this shift. Then $\{a\otimes\id,K+x\}''$ conjugated by $\ee^{-\ii pK}$ becomes $\{\sigma_p(a),x\}''$, which is \eqref{eq:frame2}.\par
\textbf{\ref{ex:crossed:3}.} $x$ commutes with $a\otimes\id$ and with $K$ (they act on different factors), hence with $K+x$. For $a'\in\M'$ write $\ee^{\ii pK}=\ee^{\ii pK_\M}\ee^{-\ii pK_{\M'}}$; $K_\M\in\M$ commutes with $a'$, so $\ee^{\ii pK}a'\ee^{-\ii pK}=\ee^{-\ii pK_{\M'}}a'\ee^{\ii pK_{\M'}}$, built from $\M'$ and $p$ and hence commuting with $\M\otimes\id$. Finally $[\ee^{\ii pK}a'\ee^{-\ii pK},K+x]=\ee^{\ii pK}[a',\ee^{-\ii pK}(K+x)\ee^{\ii pK}]\ee^{-\ii pK}=\ee^{\ii pK}[a',x]\ee^{-\ii pK}=0$.\par
\textbf{\ref{ex:crossed:4}.} With $\Omega=\sqrt p\ket{\uparrow\bar\uparrow}+\sqrt{1-p}\ket{\downarrow\bar\downarrow}$: $K_\M=\mathrm{diag}(-\log p,-\log(1-p))\otimes\id$, $K_{\M'}=\id\otimes\mathrm{diag}(-\log p,-\log(1-p))$, $K=K_\M-K_{\M'}$, and $y=x-K_{\M'}$ (acts on the sector $\ket{i\bar j}$ as $x+\log p_j$). Since $K+x=K_\M+y$ with $K_\M\in\M$, $N=\{\M,K+x\}''=\{\M,y\}''\cong M_2\otimes\Linf(\R_y)$ and the centre is $\id\otimes\Linf(\R_y)$.\par
\textbf{\ref{ex:crossed:5}.} $\ee^{-\ii tX}a\,\ee^{\ii tX}=\ee^{-\ii tH}a\,\ee^{\ii tH}$ for $a\in\M$ means $\ee^{\ii tH}\ee^{-\ii tX}$ commutes with $\M$; its generator $X-H=y$ therefore commutes with $\M$. Since $H\in\M$ is invariant under its own flow, $[X,H]=0$ and so $[y,X]=0$. Hence $N=\{\M,X\}''=\{\M,H+y\}''=\{\M,y\}''\cong\M\bar\otimes\Linf(\R_y)$.\par

\subsection*{\S\ref{ch:takesaki} Takesaki duality}
\textbf{\ref{ex:takesaki:1}.} $\theta_s(a\otimes\id)=a\otimes\id$ and $\theta_s(K+x)=K+x+s$, so $\theta_s(\ee^{\ii uX})=\ee^{\ii us}\ee^{\ii uX}$. For $b=\int\dd u\,a(u)\ee^{\ii uX}$, $\theta_sb=\int\dd u\,\ee^{\ii us}a(u)\ee^{\ii uX}$, which equals $b$ for all $s$ iff $a(u)=0$ for $u\ne0$, i.e.\ $b=a(0)\in\M\otimes\id$.\par
\textbf{\ref{ex:takesaki:2}.} (a) $\tau(FG)=\int\dd y\,\ee^y\Tr(F(y)G(y))=\tau(GF)$ by cyclicity of $\Tr$ at each $y$. (b) With $\theta_s:f(y)\mapsto f(y+s)$, $\tau(\theta_sF)=\int\dd y\,\ee^y\Tr F(y+s)=\ee^{-s}\tau(F)$. (c) $X=K_\M+y$ with $K_\M=-\log\rho$, so $f(X)=\sum_i\ket i\!\bra i f(y-\log p_i)$ and $\tau(f(X))=\sum_i\int\dd y\,\ee^yf(y-\log p_i)=\sum_ip_i\int\dd X\,\ee^Xf(X)=\int\ee^Xf(X)\dd X$.\par
\textbf{\ref{ex:takesaki:3}.} $\tau(\chi(X\le c))=\int_{-\infty}^c\ee^X\dd X=\ee^c$, so $\tau(\chi(X\le0))=1$, $\tau(\chi(X\ge0))=\int_0^\infty\ee^X\dd X=\infty$, and $c\in\R$ gives every value in $(0,\infty)$. $N$ is therefore semifinite with projections of every dimension: type II.\par
\textbf{\ref{ex:takesaki:4}.} For $\varphi_h=\Tr(h^{1/2}\rho\,h^{1/2}\cdot)=\Tr(h\rho\,\cdot)$ with $h$ commuting with $\rho$, the modular group is $\mathrm{Ad}(h\rho)^{it}=\mathrm{Ad}\,h^{it}\circ\sigma^\varphi_t$. It is trivial iff $(h\rho)^{it}$ is a scalar for all $t$, i.e.\ $h\rho\propto\id$, $h\propto\rho^{-1}$.\par

\subsection*{\S\ref{ch:entropy_II} Entropy in type II}
\textbf{\ref{ex:entropy_II:1}.} With $\hat\rho_i=\ee^{-y}p_ig(y-\log p_i)$, $\ee^y\hat\rho_i=p_ig(y-\log p_i)$ and $\log\hat\rho_i=-y+\log p_i+\log g(y-\log p_i)$. Substituting $x=y-\log p_i$: $S=-\sum_ip_i\int\dd x\,g(x)[-x-\log p_i+\log p_i+\log g(x)]=\langle x\rangle+h(g)$, independent of $p_i$. \texttt{check\_toy.py} confirms this numerically.\par
\textbf{\ref{ex:entropy_II:2}.} For a Gaussian $h(g)=\frac12\log(2\pi\ee s^2)$ and $\langle x\rangle=\mu$, so $S=\mu+\frac12\log(2\pi\ee s^2)$, which vanishes at $\mu=-\frac12\log(2\pi\ee s^2)$ and is positive for larger $\mu$. A Gaussian has support on $x>0$, outside the corner $x\le0$ where $g_{\max}$ lives and where the formula $S=-D_{\rm KL}\le0$ applies; such states are not in $P\A_{\rm cr}P$.\par
\textbf{\ref{ex:entropy_II:3}.} For $g$ supported on $x\le0$, $S=\langle x\rangle+h(g)=-\int g\log\frac g{\ee^{x}}=-D_{\rm KL}(g\|g_{\max})\le0$ with equality iff $g=g_{\max}$. For $g=\lambda\ee^{\lambda x}\theta(-x)$: $\langle x\rangle=-1/\lambda$, $h(g)=1-\log\lambda$, so $S=1-\frac1\lambda-\log\lambda$; $S'(\lambda)=\frac1{\lambda^2}-\frac1\lambda=0$ at $\lambda=1$, $S(1)=0$ and $S''<0$.\par
\textbf{\ref{ex:entropy_II:4}.} If $\tau'=c\tau$ then $\hat\Phi(F)=\tau'(\hat\rho'F)$ requires $\hat\rho'=\hat\rho/c$, and $S'=-\tau'(\hat\rho'\log\hat\rho')=-\tau(\hat\rho(\log\hat\rho-\log c))=S+\log c$. The normalization $\tau(P)=1$ fixes $c=1$ so that the maximum entropy is $0$; with $\tau'(P)=c$ it is $\log c$.\par

\subsection*{\S\ref{ch:toy_clock} Toy models: a system plus a clock}
\textbf{\ref{ex:toy_clock:1}.} $\widehat{\ket1\!\bra0}=\ket1\!\bra0\,\ee^{-\ii cp}$ with $c=\beta\omega_0$. $[K,\ket1\!\bra0]=\beta\omega_0\ket1\!\bra0$ and, from $\ee^{\ii cp}x\ee^{-\ii cp}=x+c$, $[x,\ee^{-\ii cp}]=c\,\ee^{-\ii cp}$. Hence $[K-x,\widehat{\ket1\!\bra0}]=(\beta\omega_0-c)\widehat{\ket1\!\bra0}=0$, whereas $[K-x,\ket1\!\bra0\otimes\id]=\beta\omega_0\ket1\!\bra0\ne0$.\par
\textbf{\ref{ex:toy_clock:2}.} Generators of $N$: the dressed matrix units $\widehat{e_{ij}}=e_{ij}\ee^{-\ii\beta(E_i-E_j)p}$ and $x$. In frame 1 they become $e_{ij}\otimes\id$ and $X=K+x$, and $y=X-K_\M=x-K_{\M'}$ is the central element, acting on $\ket{i\bar j}$ as $x+\log p_j$. The matrix units $e_{ij}=\ket i\!\bra j\otimes\id\in\M$ commute with $y$, giving $N\cong M_2\otimes\Linf(\R_y)$.\par
\textbf{\ref{ex:toy_clock:3}.} With $\lambda$ a multiplier for $\int f=1$, the stationarity condition of $-\beta\langle q\rangle+h(f)$ is $-\beta q-\log f-1-\lambda=0$, so $f\propto\ee^{-\beta q}$ and, normalized on $q\ge0$, $f=\beta\ee^{-\beta q}$. Then $\langle q\rangle=1/\beta$, $h(f)=1-\log\beta$ and $S=-1+(1-\log\beta)+\log\beta=0$. $\langle q\rangle=T$ is the equipartition energy $k_BT$ of a classical mode (one quadratic degree of freedom in $q$, linear energy).\par
\textbf{\ref{ex:toy_clock:4}.} The dressed matrix units are $\widehat{\ket i\!\bra j}=\ket i\!\bra j\,\ee^{-\ii\beta(E_i-E_j)p}$ for $i,j\in\{0,1,2\}$: nine operators, of which the three diagonal ones ($E_i=E_j$) carry no shift and the others shift $x$ by $\beta(E_i-E_j)$, i.e.\ the clock energy $q$ by $-(E_i-E_j)$. The entropy formula $S=\langle x\rangle+h(g)$ of \cref{ch:entropy_II} holds for any number of levels (the $p_i$ drop out as in the qubit case), so $S(\Omega\otimes g^{1/2})$ is unchanged.\par
\textbf{\ref{ex:toy_clock:5}.} On a grid $x\in[-40,0]$, \texttt{dx=x[1]-x[0]; g=np.exp(x)}: \texttt{g.sum()*dx} $\to1$ and \texttt{(-g*np.log(g)).sum()*dx+(x*g).sum()*dx} $\to0$ (the two terms are $+\!\int|x|\ee^x=1$ and $-1$). For any other density supported on $x\le0$, $S=-D_{\rm KL}(g\|\ee^{x}\theta(-x))<0$; e.g.\ $g=\lambda\ee^{\lambda x}$ with $\lambda=2$ gives $1-\frac12-\log2\approx-0.193$.\par

\section*{Chapter~\ref{ch:aqft}: Algebraic quantum field theory}

\subsection*{\S\ref{ch:haagkastler} Haag--Kastler axioms}
\textbf{\ref{ex:haagkastler:1}.} With $W(f)=\ee^{\ii\phi(f)}$ one has $W(f)W(g)=\ee^{-\ii\sigma(f,g)}W(g)W(f)$ with $\sigma(f,g)=\int\!\!\int f(x)\Delta(x-y)g(y)$, where $\Delta$ is the Pauli--Jordan commutator function. $\Delta(x-y)=0$ for spacelike $x-y$, so $\sigma=0$ when the supports are spacelike separated and the Weyl operators, which generate $\A(O_1)$ and $\A(O_2)$, commute.\par
\textbf{\ref{ex:haagkastler:2}.} If $\A(O)=B(K)\otimes\id$ and $\A(O')=\A(O)'=\id\otimes B(K')$ then the vacuum $\Omega\in K\otimes K'$ would have a Schmidt decomposition and a density matrix $\rho$, so its modular operator $\Delta=\rho\otimes\rho'^{-1}$ would have pure point spectrum. For a wedge the modular operator is $\ee^{-2\pi B}$ with continuous spectrum (Bisognano--Wichmann), so this is impossible: $\A(O)$ is not type I. Equivalently, the entanglement across $\partial O$ has no finite density-matrix description (\cref{ch:typeIII_local}).\par
\textbf{\ref{ex:haagkastler:3}.} $\A(I)=\{W(f):\mathrm{supp}f\subset I\}''$, $W(f)=\ee^{\ii\phi(f)}$ with $[\phi(f),\phi(g)]=\frac\ii2\int f g'$. Translations $U(a)$ and dilations $D(\lambda)$ act as $U(a)\phi(f)U(a)^*=\phi(f_a)$, $f_a(x)=f(x-a)$, and $D(\lambda)\phi(f)D(\lambda)^*=\phi(f_\lambda)$, $f_\lambda(x)=f(x/\lambda)$ (with the weight of the current), mapping $\A(I)\to\A(I+a)$ and $\A(\lambda I)$: the action is geometric.\par

\subsection*{\S\ref{ch:rs_bw} Reeh--Schlieder, Bisognano--Wichmann, and the Unruh effect}
\textbf{\ref{ex:rs_bw:1}.} The chiral fields depend only on $x^\pm=t\pm x$. A boost with rapidity $s$ rescales $x^+\to\ee^{s}x^+$ and $x^-\to\ee^{-s}x^-$, so on one-particle states it rescales the null momentum $p\mapsto\ee^{\mp s}p$. The modular operator is $\Delta^{1/2}=\ee^{-\pi B}$, i.e.\ the boost continued to $s=-\ii\pi$, which sends $p\mapsto\ee^{\ii\pi}p=-p$: positive-momentum wave packets supported on the half-line map to analytic continuations that are boundary values in the upper half $p$-plane (a Hardy-space statement), which is why $\Delta^{1/2}$ can map the right-wedge one-particle space onto the left one.\par
\textbf{\ref{ex:rs_bw:2}.} $K\Omega=0$ so $\langle\Omega|K|\Omega\rangle=0$. For $\Psi=u\Omega$ with $u\in\A(W)$ one has $\Delta_{\Omega,\Psi}=\Delta_\Omega$ (\cref{ch:modham}), so $S_{\rm rel}(\Psi\|\Omega)=-\langle\Psi|\log\Delta_{\Omega,\Psi}|\Psi\rangle=\langle\Psi|K|\Psi\rangle=2\pi\langle\Psi|B|\Psi\rangle\ge0$ with equality iff $\Psi=\Omega$. This avoids needing pointwise positivity of the boost energy density.\par
\textbf{\ref{ex:rs_bw:3}.} The modular flow is a boost of rapidity $-2\pi t$ with period $2\pi$ in rapidity for the KMS condition ($\beta=2\pi$). A uniformly accelerated observer has rapidity $\eta=a\tau/c$, so the thermal period in proper time is $2\pi c/a$ and $T=\frac{1}{k_B}\frac{\hbar a}{2\pi c}$, i.e.\ $T_U=\hbar a/(2\pi ck_B)$, which is $a/2\pi$ in natural units.\par
\textbf{\ref{ex:rs_bw:4}.} By Reeh--Schlieder $\Omega$ is cyclic for $\A(O')$, hence separating for $\A(O')'=\A(O)$, so the vacuum state is faithful on $\A(O)$. If it were pure and normal, its support projection (which is $\id$ by faithfulness) would be minimal, so $\A(O)=\C\id$-corner of a minimal projection $=B(\C)$-type; more precisely a pure normal state forces a minimal projection and therefore $\A(O)$ would be type I, contradicting the type III property of \cref{ch:typeIII_local}.\par

\subsection*{\S\ref{ch:typeIII_local} Local algebras are type III$_1$}
\textbf{\ref{ex:typeIII_local:1}.} A type I$_\infty$ factor $B(K)\otimes\id$ has minimal projections and hence pure normal states $\omega_\varphi$, $\varphi\in K$. It also forces the vacuum to have a density matrix $\rho$ on $K$ and $\Delta=\rho\otimes\rho'^{-1}$ with pure point spectrum. Cyclicity of the vacuum for $\A(O')$ (Reeh--Schlieder) makes $\Omega$ separating for $\A(O)$, which is consistent with a faithful $\rho$; the contradiction comes from the modular operator: for the wedge, $\Delta=\ee^{-2\pi B}$ has continuous spectrum, so $\A(O)$ cannot be of this form.\par
\textbf{\ref{ex:typeIII_local:2}.} The regularized entropy is $\frac13\log L-\frac13\log\epsilon$. One can subtract $-\frac13\log\epsilon$, but that depends on the cutoff scheme: a different regularization (cutoff shape or lattice) changes the finite part by a constant that depends on the state only through the choice of scheme, and no covariant continuum prescription fixes it. In type III there is no density matrix, so $S$ is not a function on the state space at all; only relative quantities such as $S_{\rm rel}$ and differences of entropies at fixed cutoff survive.\par
\textbf{\ref{ex:typeIII_local:3}.} Wedge of a free field: type III$_1$ (Bisognano--Wichmann; modular spectrum $(0,\infty)$). Double cone in a CFT: type III$_1$ (conformally equivalent to a wedge). Whole Minkowski space in the vacuum representation: $\A=B(\HH)$ (irreducible), type I$_\infty$, not III. Quantum-mechanical particle: $B(L^2(\R))$, type I$_\infty$.\par

\subsection*{\S\ref{ch:dhr} Superselection and DHR}
\textbf{\ref{ex:dhr:1}.} $\rho(W(f))W(g)=\ee^{\ii\lambda\ell(f)}W(f)W(g)$ and $\ell$ linear gives $\rho(W(f)W(g))=\rho(W(f))\rho(W(g))$, so $\rho$ is a $^*$-endomorphism (indeed an automorphism). If $\mathrm{supp}f\subset O'$ then $\ell(f)=0$, so $\rho(W(f))=W(f)$ and $\rho$ acts trivially on $\A(O')$: it is DHR-localized in $O$. It shifts the field, $\phi\mapsto\phi+\lambda\ell$, i.e.\ it is a coherent shift by a classical background; if $\ell=\sigma(h,\cdot)$ with $h$ supported in $O$ then $\rho=\mathrm{Ad}\,W(\lambda h)$ is inner, hence a trivial sector.\par
\textbf{\ref{ex:dhr:2}.} Define the grading automorphism $\gamma$ by $\gamma(\psi)=-\psi$ on fermion fields and $\gamma=\id$ on bosons; it is implemented by $(-1)^F$, which commutes with every bosonic (even) operator and anticommutes with fermionic ones. Observables are the even operators, which commute with $(-1)^F$, so the Hilbert space splits into the $\pm1$ eigenspaces of $(-1)^F$ (even and odd fermion number): the two irreducible representations of the gauge group $\Z_2$ are the two superselection sectors.\par

\section*{Chapter~\ref{ch:gravity_ds}: Gravity, constraints, and de Sitter space}

\subsection*{\S\ref{ch:gravity_constraints} Gravity as constraints}
\textbf{\ref{ex:gravity_constraints:1}.} With $e^{\ii\lambda T}H_C\ee^{-\ii\lambda T}=H_C+\lambda$ we have $\ee^{-\ii\lambda T}H_C\ee^{\ii\lambda T}=H_C-\lambda$. Let $U=\ee^{\ii H_ST}$; $H_S$ commutes with $T$ and $H_C$, so $U^{-1}(H_S+H_C)U=H_S+H_C-H_S=H_C$, and $[H_C,a]=0$ since $a$ acts on the system, giving $[H_S+H_C,UaU^{-1}]=U[H_C,a]U^{-1}=0$. If instead $[T,H_C]=+\ii$ the same calculation gives $U^{-1}(H_S+H_C)U=2H_S+H_C$ and $a(T)$ is \emph{not} invariant; the invariant dressing is then $\ee^{-\ii H_ST}a\,\ee^{\ii H_ST}$ (equivalently, the clock reading runs backwards).\par
\textbf{\ref{ex:gravity_constraints:2}.} Invariant operators commute with $H=\frac{p^2}{2m}+H_C$. As in \cref{ch:crossed}, since the flow generated by $p^2/2m$ is inner in $\Bnd{\HH_S}$ the invariant algebra is generated by the dressed operators $\ee^{-\ii H_ST}a\,\ee^{\ii H_ST}$ and $H_C$, i.e.\ $\Bnd{\HH_S}\otimes\Linf(\R_{H_C})$. Physical states are not normalizable solutions of $H\Psi=0$; group averaging $\int\dd t\,\ee^{-\ii tH}$ defines the inner product $\langle\!\langle\Psi,\Phi\rangle\!\rangle=\langle\Psi,2\pi\delta(H)\Phi\rangle$, which makes the physical Hilbert space $\cong\HH_S$ (the clock being used up), equivalently the coinvariants $\HH/H\HH$.\par
\textbf{\ref{ex:gravity_constraints:3}.} In a closed universe the constraints generate diffeomorphisms of the whole space, so a gauge-invariant operator must be invariant under moving the point: the local field $\phi(x)$ at a fixed coordinate $x$ changes under a diffeomorphism and cannot be an observable. Dressing $\phi$ relative to a physical reference (the position of an observer, a clock, or the value of other fields) defines $\phi$ ``at the point where the reference is'', which is invariant because the reference moves with the diffeomorphism.\par

\subsection*{\S\ref{ch:gauge_warmup} Gauge theory warm-up: edge modes and centres}
\textbf{\ref{ex:gauge_warmup:1}.} Take the plaquette cut through one link $\ell$. Gauge-invariant operators of $A$: products of $Z$'s along closed loops inside $A$, Wilson lines ending on the cut link, and the electric operator $X_\ell$ on the cut link, which commutes with everything in $A$ and with everything in $B$ and is therefore central. Take $\Psi=\frac1{\sqrt2}(\ket{+}_\ell\ket{\psi_+}+\ket{-}_\ell\ket{\psi_-})$ with $p_\pm=\frac12$: then $S=-\sum p_e\log p_e+\sum p_eS^{(e)}=\log2+\frac12(S^{(+)}+S^{(-)})$, the first term being the classical (centre) entropy and the rest the entropy within each superselection sector.\par
\textbf{\ref{ex:gauge_warmup:2}.} $X_\ell$ ($\ell\in\Sigma$) commutes with all gauge-invariant operators of $A$ (they do not change the electric flux on the cut links on their own: Wilson lines ending on the cut are dressed by the matching flux operator) and with those of $B$, so it is central in $\A_A$. In the full lattice algebra there are Wilson lines crossing the cut that change $e_\ell$, so $X_\ell$ no longer commutes with everything and is not central.\par
\textbf{\ref{ex:gauge_warmup:3}.} The central elements $X_\ell$ take definite values $e_\ell$ in each irreducible representation of $\A_A$, so the spectrum of the centre labels a direct-sum decomposition $\A_A=\bigoplus_e\A_A^{(e)}$. A factor has trivial centre; in each sector $e$ one has a factor, and $e$ is the superselection (flux) sector label of \cref{ch:factors}.\par

\subsection*{\S\ref{ch:ds_special} Why de Sitter is special}
\textbf{\ref{ex:ds_special:1}.} Near $r=\ell$ write $f(r)=1-r^2/\ell^2\approx\frac2\ell(\ell-r)$; with $\rho$ the proper radial distance, $f\approx\rho^2/\ell^2$ and the metric $f\,\dd\tau^2+\dd\rho^2$ is smooth iff $\tau/\ell$ has period $2\pi$, i.e.\ $\tau\sim\tau+2\pi\ell$. Then $T_{\rm dS}=1/(2\pi\ell)$ and for $d+1=4$, $S_{\rm dS}=\frac A{4G}=\frac{\pi\ell^2}{G_N}$.\par
\textbf{\ref{ex:ds_special:2}.} $f(r)=1-\frac{2Gm}r-\frac{r^2}{\ell^2}$. At $r=\ell-\delta$: $f\approx-\frac{2Gm}\ell+\frac{2\delta}\ell=0$, so $\delta=Gm$, $r_c=\ell-Gm$. $A=4\pi r_c^2\approx4\pi\ell^2-8\pi\ell Gm$, so $\delta(A/4G)=-2\pi\ell\,m=-\beta_{\rm dS}\delta E$ with $E=m$.\par
\textbf{\ref{ex:ds_special:3}.} On a closed spatial slice the ADM boundary term in $\int N\mathcal H$ vanishes (there is no boundary at infinity), so the full generator is a pure constraint and $\int_{S^d}N\mathcal H\approx0$ generates gauge transformations only. In asymptotically flat space $\int N\mathcal H$ must be supplemented by the boundary term $N E_{\rm ADM}$ at infinity for the variation to be well defined, and that term is the physical energy, which is not gauge.\par

\subsection*{\S\ref{ch:static_patch} QFT in the static patch}
\textbf{\ref{ex:static_patch:1}.} For the conformally coupled scalar in $\mathrm{dS}_4$ the Wightman function along $r=0$ is $G(t)=-\frac1{16\pi^2\ell^2\sinh^2((t-\ii\epsilon)/2\ell)}$. Since $\sinh^2(\frac{t-2\pi\ii\ell}{2\ell})=\sinh^2(\frac t{2\ell}-\ii\pi)=\sinh^2(\frac t{2\ell})$, $G$ is periodic in imaginary time with period $2\pi\ell$, and $G(t-\ii\beta)=G(-t)$, which is the KMS condition at $\beta=2\pi\ell$ (the same follows in $\mathrm{dS}_2$ from the logarithmic two-point function).\par
\textbf{\ref{ex:static_patch:2}.} $\langle H_R\rangle$ and $\langle H_L\rangle$ are expectation values of the static Hamiltonian of each patch in the thermal state at $T_{\rm dS}$; the number of modes near the horizon is unbounded, so each diverges (UV, from the blueshift at the horizon). By the reflection symmetry exchanging the two patches they are equal, so $\langle H\rangle=\langle H_R\rangle-\langle H_L\rangle=0$, as it must since $H\Psi_{\rm dS}=0$.\par
\textbf{\ref{ex:static_patch:3}.} Dictionary: boost generator $B\leftrightarrow$ static Hamiltonian $H$ (with $K=\beta_{\rm dS}H=2\pi\ell H$ playing $2\pi B$), $2\pi$ in rapidity $\leftrightarrow\beta_{\rm dS}=2\pi\ell$ in static time, acceleration $a\leftrightarrow1/\ell$ (surface gravity of the horizon), Unruh temperature $a/2\pi\leftrightarrow T_{\rm dS}=1/(2\pi\ell)$. The Poincar\'e group is replaced by the de Sitter group $SO(d+1,1)$, of which the static patch preserves $\R_t\times SO(d)$.\par

\subsection*{\S\ref{ch:add_observer} Adding an observer}
\textbf{\ref{ex:add_observer:1}.} $\ee^{-\ii t(K-x)}=\ee^{-\ii tK}\ee^{\ii tx}$ because $K$ and $x$ commute. Then $\ee^{\ii tx}\ee^{-\ii pK}\ee^{-\ii tx}=\ee^{-\ii(p-t)K}$ using $\ee^{\ii tx}p\,\ee^{-\ii tx}=p-t$, and so $\ee^{-\ii t(K-x)}\sigma_p(a)\ee^{\ii t(K-x)}=\ee^{-\ii tK}\ee^{-\ii(p-t)K}a\,\ee^{\ii(p-t)K}\ee^{\ii tK}=\ee^{-\ii pK}a\,\ee^{\ii pK}=\sigma_p(a)$. Differentiating at $t=0$ gives $[K-x,\sigma_p(a)]=0$.\par
\textbf{\ref{ex:add_observer:2}.} $\ee^{-\ii pK}\phi(s)\ee^{\ii pK}$ is the field translated along the worldline by the modular flow, i.e.\ by static time of order $\beta_{\rm dS}p$ (sign set by the convention $\sigma_t=\mathrm{Ad}\,\Delta^{\ii t}$). So $\hat\phi(f)=\int\dd s\,f(s)\phi(s\mp\beta_{\rm dS}p)$ is the field smeared along the worldline and read off at the clock value $p$: the operator ``measure $\phi(f)$ when the clock reads $p$''; it commutes with the constraint because the clock reading $p$ is conjugate to the generator it compensates.\par
\textbf{\ref{ex:add_observer:3}.} Writing $\Psi(q)=\Psi_{-q}\otimes|q\rangle$, the equation $(H+q)\Psi=0$ forces the support of the $H$-spectral measure to sit on $H=-q$, giving a delta function in $q$, so $\|\Psi\|^2=\int\dd q\,\delta(0)\ldots=\infty$: the continuous spectrum of $H+q$ has no normalizable zero eigenvectors. Group averaging replaces $\Psi$ by $\int\dd t\,\ee^{-\ii t(H+q)}\Psi=2\pi\delta(H+q)\Psi$ and defines $\langle\!\langle\Psi,\Phi\rangle\!\rangle=\langle\Psi,2\pi\delta(H+q)\Phi\rangle$, finite on a dense set; compare the unitary group averaging of \cref{ch:groups} for non-compact groups, where the invariant subspace is empty but the averaged inner product is well defined.\par

\section*{Chapter~\ref{ch:clpw_ch}: The CLPW algebra}

\subsection*{\S\ref{ch:clpw} The CLPW algebra}
\textbf{\ref{ex:clpw:1}.} $\tau(P)=\int_{-\infty}^0\ee^{x}\dd x=1$ and $\tau(\chi(x\le c))=\int_{-\infty}^c\ee^x\dd x=\ee^c$. The corner $\chi(x\le c)$ means $q\ge q_0=-c/\beta_{\rm dS}$, so $\tau=\ee^{-\beta_{\rm dS}q_0}$, the fraction of the full algebra's ``dimension'' left when the observer has energy at least $q_0$ (the Boltzmann factor).\par
\textbf{\ref{ex:clpw:2}.} For a density matrix $\rho$ of $P\A_{\rm cr}P$ with $\tau(\rho)=1$, $S=-\tau(\rho\log\rho)=-\tau(P\,\rho\log\rho)$, and $u\mapsto u\log u$ is convex, so by Jensen $\tau(\rho\log\rho)\ge\tau(\rho)\log\tau(\rho)=0$ with $\tau(P)=1$ as the probability measure. Hence $S\le0$, with equality iff $\rho$ is constant on the spectrum, i.e.\ $\rho=P$.\par
\textbf{\ref{ex:clpw:3}.} With $\tau=\Tr/N$ the density matrix is $\rho_\tau=N\rho$, so $S_\tau=-\Tr(\rho\log(N\rho))=S_{\rm vN}-\log N\le0$ since $S_{\rm vN}\le\log N$. For $N=\ee^{S_{\rm dS}}$ the CLPW entropy is $S_{\rm vN}-S_{\rm dS}$, the amount by which the von Neumann entropy of the ``qubit'' Hilbert space of $\ee^{S_{\rm dS}}$ states falls short of the maximum.\par
\textbf{\ref{ex:clpw:4}.} For a projection $\Lambda$ the probability in $\hat\rho=\id$ is $\tau(\Lambda)$ and $\rho_\Lambda=\Lambda/\tau(\Lambda)$ has $S=\log\tau(\Lambda)$ (since $-\tau(\rho_\Lambda\log\rho_\Lambda)=\log\tau(\Lambda)$), so $p_\Lambda=\tau(\Lambda)=\ee^{\Delta S_\Lambda}$. For $\Lambda=\chi(q\ge q_0)$: $\ee^{-\beta_{\rm dS}q_0}$. And $\tau(\chi(q_1\le q\le q_2))=\ee^{-\beta_{\rm dS}q_1}-\ee^{-\beta_{\rm dS}q_2}$, which is exactly the continuous dimension $d$ of $\Pi'_{[q_1,q_2]}$ in \cref{ch:clpw}.\par
\textbf{\ref{ex:clpw:5}.} For $q\ge m$, $\tau(\id)=\int_{-\infty}^{-\beta_{\rm dS}m}\ee^x\dd x=\ee^{-\beta_{\rm dS}m}$. To normalize, multiply $\tau$ by $c=\ee^{\beta_{\rm dS}m}$; then $S\to S+\log c=S+\beta_{\rm dS}m$.\par
\textbf{\ref{ex:clpw:6}.} For the exponential density on $x\le0$, $\mathrm{Var}(x)=1$, so $\Delta x=1$ and $\Delta p\ge\frac12$ (in $[x,p]=\ii$ units). The clock time is $\beta_{\rm dS}p$, so its resolution is $\gtrsim\beta_{\rm dS}/2$, comparable to $\beta_{\rm dS}$, not $\ll\beta_{\rm dS}$ as required by \eqref{eq:semiclass}. So $\Psi_{\max}$ is not semiclassical.\par

\subsection*{\S\ref{ch:anatomy} Anatomy of the result}
\textbf{\ref{ex:anatomy:1}.} $\langle x\rangle=-1/\lambda$, $h(g)=-\int g\log g=-\log\lambda+1$, so $S=1-\frac1\lambda-\log\lambda$ (maximal at $\lambda=1$). In the variable $q$ the density is $\propto\ee^{-\lambda\beta_{\rm dS}q}$: a Boltzmann law at inverse temperature $\lambda\beta_{\rm dS}$, i.e.\ temperature $T_{\rm dS}/\lambda$.\par
\textbf{\ref{ex:anatomy:2}.} At $r=\ell-\delta$: $f=1-\frac{2G_NE}\ell-(1-\frac{2\delta}\ell)+\mathcal O(\delta^2,E\delta)=\frac{2\delta}\ell-\frac{2G_NE}\ell=0$, so $\delta=G_NE$. $A=4\pi r_{\rm hor}^2=4\pi\ell^2-8\pi\ell G_NE+\ldots$ and $\delta(A/4G_N)=-2\pi\ell E=-\beta_{\rm dS}E$.\par
\textbf{\ref{ex:anatomy:3}.} The cocycle identity $h_{\Phi,\Psi}-h_\Psi=h_\Phi-h_{\Psi,\Phi}$ and $\langle\Phi|h_\Phi|\Phi\rangle=0$ give $\langle h_{\Phi,\Psi}\rangle=\langle K\rangle-\langle h_{\Psi,\Phi}\rangle=\langle K\rangle-S_{\rm rel}(\Phi\|\Psi)$. In type I, $-\log\Delta_{\Phi,\Psi}=-\log\rho_\Phi\otimes\id+\id\otimes\log\sigma'_\Psi$, so $\langle h_{\Phi,\Psi}\rangle=S(\rho_\Phi)+\Tr\rho'_\Phi\log\sigma'_\Psi$. Meanwhile $\langle K\rangle=-\Tr\rho_\Phi\log\rho_\Psi+\Tr\rho'_\Phi\log\sigma'_\Psi$ and $S_{\rm rel}=\Tr\rho_\Phi(\log\rho_\Phi-\log\rho_\Psi)$, whose difference is the same expression.\par
\textbf{\ref{ex:anatomy:4}.} The differences are about $0.22,0.12,0.04,0.011,0.0028,7\times10^{-4},1.7\times10^{-4}$ for $s=0.5,\dots,32$, dropping by $\approx4$ per doubling of $s$: $\propto s^{-2}$. For $\Phi=u\Omega$, $\Delta_{\Omega,\Phi}=\Delta_\Omega$ exactly, so $S_{\rm rel}=\langle K\rangle$ and the formula reduces to $\langle x\rangle+h(g)$, which is the exact entropy (\cref{ch:entropy_II}): no approximation is made, so the discrepancy vanishes for every $g$.\par
\textbf{\ref{ex:anatomy:5}.} \eqref{eq:wall} gives $\Sgen(B)=\Sgen(\infty)-S_{\rm rel}$. Once the matter has left the patch, $\Sgen(\infty)$ is the generalized entropy of empty de Sitter space plus observer, i.e.\ the first-law area $A_{\rm dS}/4G_N-\beta_{\rm dS}\langle H_{\rm obs}\rangle$ plus the observer's entropy. So $\Sgen(B)-\Sgen(\text{empty dS+observer})=-S_{\rm rel}\le0$.\par
\textbf{\ref{ex:anatomy:6}.} The semiclassical formula was derived for slowly varying $g$ ($\epsilon\ll1$), which $g_{\max}$ is not; the clock resolution of $\Psi_{\max}$ is $\gtrsim\beta_{\rm dS}/2$. There is no contradiction: for $\Phi=\Psi_{\rm dS}$ (i.e.\ $u=\id$) the formula is exact for every $g$, and gives $h(g_{\max})+\langle x\rangle=1-1=0$, in agreement with $S(\Psi_{\max})=0$. What fails outside the semiclassical regime is only the interpretation via Wall's formula and the first law, and the control over general $\Phi$ not of the form $u\Omega$.\par

\subsection*{\S\ref{ch:qubit} A worked example: a qubit system with a clock}
\textbf{\ref{ex:qubit:1}.} For $F=\sum_{ab}\ket a\!\bra b\otimes f_{ab}(y)$, $\hat\Phi(F)=\sum_{a,b,j}\int\dd x\,\bar c_{aj}c_{bj}f_{ab}(x+\log p_j)g(x)$. Substituting $y=x+\log p_j$ gives $\int\dd y\sum_{ab}f_{ab}(y)\sum_j\bar c_{aj}c_{bj}g(y-\log p_j)$. Equating with $\tau(\hat\rho F)=\int\dd y\,\ee^y\sum_{ab}\hat\rho_{ba}(y)f_{ab}(y)$ gives $\hat\rho_{ba}(y)=\ee^{-y}\sum_j\bar c_{aj}c_{bj}g(y-\log p_j)$, i.e.\ \eqref{eq:rhoqubit}.\par
\textbf{\ref{ex:qubit:2}.} For $\Phi=\ket{bb}$ the proposition gives $S=h(g)+\langle x\rangle+\log p_b$. So $S_{11}-S_{00}=\log p_1-\log p_0=\log(p_1/p_0)$; with $p_b=\ee^{-\beta E_b}/Z$ this equals $-\beta(E_1-E_0)$.\par
\textbf{\ref{ex:qubit:3}.} The Schmidt vectors are $\cos\theta\ket0$ and $\sin\theta\ket1$ (orthogonal), so $q_0=\cos^2\theta$, $q_1=\sin^2\theta$ and $S=h(g)+\langle x\rangle-\sum_bq_b\log(q_b/p_b)$. For fixed $g$ this is maximal when the relative entropy vanishes, $q_b=p_b$, i.e.\ $\cos^2\theta=p_0$, which is the modular state $\Omega$ itself ($u=\id$).\par
\textbf{\ref{ex:qubit:4}.} Replace \texttt{n=3} by \texttt{n=2} and a $c$ with orthogonal columns (e.g.\ $c=\mathrm{diag}(\cos\theta,\sin\theta)$). The printed differences $S_{\rm exact}-S_{\rm CLPW}$ are of order $10^{-10}$ (floating-point integration error) for every width, in contrast to the general-$c$ case where they scale as $s^{-2}$.\par
\textbf{\ref{ex:qubit:5}.} On the sector $\ket{ab}$, $X=K+x=-\log p_a+\log p_b+x$, so $X\le0$ is $x\le\log p_a-\log p_b$. For $\Phi=\ket{00}$ this is $x\le0$ and $\langle P\rangle=\int_{-\infty}^0g$; for $\Phi=\ket{01}$ ($a=0,b=1$) it is $x\le\log(p_0/p_1)$ and $\langle P\rangle=\int_{-\infty}^{\log(p_0/p_1)}g$.\par

\section*{Chapter~\ref{ch:observers}: Observers, entropy, and open questions}

\subsection*{\S\ref{ch:clocks_qrf} Observers, clocks, and reference frames}
\textbf{\ref{ex:clocks_qrf:1}.} For a periodic clock $q\in\Z$ and $H_S$ with integer-spaced energies, $\ee^{-2\pi\ii H_S}$ is a scalar, so dressing by $\ee^{-\ii pH_S}$ is well defined for $p\in\R/2\pi\Z$. The invariants of $\Bnd{\HH_S}\otimes\Bnd{\ell^2(\Z)}$ under $H_S+q$ are generated by the dressed operators and $q$, giving $\Bnd{\HH_S}\otimes\ell^\infty(\Z)$: the same as the $\R$ case with $\Linf(\R)$ replaced by $\ell^\infty(\Z)$ (discrete clock energy). For a type III system with periodic modular flow ($\sigma_{t_0}=\mathrm{id}$, Powers factor) the $U(1)$-crossed product has a non-trivial centre, as in \cref{ch:crossed}.\par
\textbf{\ref{ex:clocks_qrf:2}.} A dressed operator $\widehat{\ket i\!\bra j}=\ket i\!\bra j\,\ee^{-\ii p(E_i-E_j)}$ shifts the clock energy $q$ by $-(E_i-E_j)$, which must be a multiple of $\delta$ for $q$ to stay in $\{0,\delta,2\delta,\dots\}$, so the transition energies lie in $\delta\Z$. If they generate only a subgroup $m\delta\Z$ then $q\bmod m\delta$ is conserved by every dressed operator and $\ee^{2\pi\ii q/m\delta}$ is central. The dressed algebra is a factor iff the transition energies generate $\delta\Z$ (no non-trivial conserved residue).\par
\textbf{\ref{ex:clocks_qrf:3}.} A clock with $\Delta q\ll T_{\rm dS}$ has time uncertainty $\gg\beta_{\rm dS}$: it does not resolve the modular flow, so it cannot be used to dress operators at definite times. In terms of \eqref{eq:semiclass} this is $\epsilon\gg1$ (the opposite of $\epsilon\ll1$), the density $g$ is sharply peaked, $h(g)\to-\infty$ logarithmically and the density-matrix expansion fails. Effectively the clock's energy is fixed, the constraint reduces to an ordinary constraint on the fields, and the entropy reduces to the von Neumann entropy of the fields (with the clock traced out) rather than the generalized entropy.\par

\subsection*{\S\ref{ch:what_observer} What counts as an observer?}
\textbf{\ref{ex:what_observer:1}.} If $[T,q]=\ii$ with both self-adjoint (in the Weyl sense) then $\ee^{\ii aT}q\,\ee^{-\ii aT}=q+\ii a[T,q]=q-a$, so $q$ and $q-a$ are unitarily equivalent for every real $a$ and $\mathrm{spec}(q)=\mathrm{spec}(q)-a$ for all $a$. A non-empty closed subset of $\R$ invariant under all translations is $\R$. Hence $q\ge0$ is incompatible with a self-adjoint $T$.\par
\textbf{\ref{ex:what_observer:2}.} Take a spin $j$, $q=J_z+j\in\{0,1,\dots,2j\}$ and the phase operator $\ee^{\ii\varphi}=\sum_{n<2j}\ket{n+1}\!\bra n+\ket0\!\bra{2j}$ (cyclic shift). The commutator $[q,\ee^{\ii\varphi}]=\ee^{\ii\varphi}$ holds on all states except those touching the wrap-around at $q=0,2j$. As $j\to\infty$, on states supported at $q\gg1$ and $2j-q\gg1$ (the regime $\epsilon\ll1$) $\varphi$ acts as an ideal conjugate time and the algebra generated by dressed operators and $q\ge0$ approaches the II$_1$ crossed-product corner of \cref{ch:clpw_ch}; for finite $j$ the clock algebra is $M_{2j+1}$ and the combined algebra is type I.\par
\textbf{\ref{ex:what_observer:3}.} With two clocks of energies $q_1,q_2\ge0$ the constraint is $H+q_1+q_2$ and the invariants of $\A\otimes B(L^2(q_1))\otimes B(L^2(q_2))$ are generated by the operators dressed to either clock, $q_1$, $q_2$; this is a crossed product of $\A$ with $\R$ augmented by a relative-time operator (the difference of the two clock readings). It is not the tensor product of the two single-observer algebras $\A_{\rm cr}^{(1)}\otimes\A_{\rm cr}^{(2)}$, because there is only one copy of $H$ to compensate: the tensor product would need two independent constraints $H+q_1$ and $H+q_2$ (two copies of the patch).\par

\subsection*{\S\ref{ch:gen_entropy} Generalized entropy and the second law}
\textbf{\ref{ex:gen_entropy:1}.} In finite dimensions take $\M_1=B(\HH_A\otimes\HH_B)$, $\M_2=B(\HH_A)\otimes\id$, and states $\rho,\sigma$. Then $\psi|_{\M_2}=\rho_A=\Tr_B\rho$ and $S_{\rm rel}(\rho_A\|\sigma_A)\le S_{\rm rel}(\rho\|\sigma)$ is data processing for the partial trace, proved in \cref{ch:monotonic} from joint convexity of relative entropy using the unitary twirl on $B$.\par
\textbf{\ref{ex:gen_entropy:2}.} For null cuts $\lambda_i$ the relative entropy is $S_{\rm rel}(\lambda_i)=2\pi\int_{\lambda_i}^\infty(\lambda-\lambda_i)T_{\lambda\lambda}\,\dd\lambda\,\dd^{d-2}x_\perp-\Delta S(\lambda_i)$. Monotonicity in $\lambda_i$ (smaller algebra for larger $\lambda_i$) gives $\partial_{\lambda_i}S_{\rm rel}\le0$, i.e.\ $2\pi\int_{\lambda_i}^\infty T\,\dd\lambda\ge-\partial_{\lambda_i}\Delta S$, an integrated QNEC-type inequality. The local QNEC, $2\pi T_{\lambda\lambda}\ge\partial_\lambda^2\Delta S$, is the stronger statement $\partial_{\lambda_i}^2S_{\rm rel}\ge0$ (convexity), which does not follow from first-order monotonicity alone and is the content of the Ceyhan--Faulkner results.\par
\textbf{\ref{ex:gen_entropy:3}.} The GSL compares $\Sgen$ at two cuts, $\Sgen(\lambda_2)-\Sgen(\lambda_1)=S(\lambda_2)-S(\lambda_1)$ in terms of the algebraic entropy. This holds only if the additive constant is the same for both cuts; if it depended on $\lambda$ the monotonicity of $S_{\rm rel}$ would not translate into monotonicity of $\Sgen$.\par

\subsection*{\S\ref{ch:open_ds} Open questions in de Sitter}
\textbf{\ref{ex:open_ds:1}.} The maximally mixed state of I$_N$ has $S=\log N$; the II$_1$ maximum is $S=0$ in the normalization $\tau(\id)=1$. Passing from I$_N$ to II$_1$ keeps only entropy \emph{differences} (and the shape of the spectrum) but loses the absolute scale $\log N$, i.e.\ the number of states: the II$_1$ trace is normalized by convention, so $\ee^{S_{\rm dS}}$ is not determined by the algebra.\par
\textbf{\ref{ex:open_ds:2}.} Without an observer the invariant algebra of a static patch is trivial, so the observer is what makes any observable exist at all; its properties (energy bounded below, mass, clock resolution, SO($d$) frame) enter the algebra and the results (e.g.\ the shift $\beta_{\rm dS}m$ of the maximal entropy and the condition $\epsilon\ll1$). Whether the entropy is independent of such details is an open question, so the observer is a physical input rather than a technical device.\par

\section*{Chapter~\ref{ch:bh}: Black holes and holography}

\subsection*{\S\ref{ch:large_N} Large $N$ and emergent type III$_1$}
\textbf{\ref{ex:large_N:1}.} For the single-trace (generalized free field) algebra in the thermal (TFD) state the modular operator is $\Delta=\ee^{-\beta(H_R-H_L)}$ and on the one-particle space it acts as $\ee^{-\beta\omega}$. At large $N$ the boundary two-point function has a continuous spectral density (quasinormal modes form a continuum in the strict large-$N$ limit), so $\omega$ ranges over $\R$ continuously and $\Delta$ has continuous spectrum $(0,\infty)$, which is type III$_1$.\par
\textbf{\ref{ex:large_N:2}.} In the TFD GNS space the modular conjugation $J$ exchanges right and left (it maps $\M_R=\{$right single-trace operators$\}''$ to the algebra generated by the left ones), and Tomita--Takesaki gives $J\M_RJ=\M_R'$. The left single-trace operators commute with the right ones and generate $\M_L=J\M_RJ$, so $\M_R'=\M_L$.\par

\subsection*{\S\ref{ch:bh_crossed} Crossed product for black holes}
\textbf{\ref{ex:bh_crossed:1}.} By \cref{ch:takesaki} the trace on $\M\rtimes_\sigma\R$ is $\tau(\hat a)=\int\dd x\,\ee^{x}\langle\Psi|a(x)|\Psi\rangle$ with $x$ the clock variable, which for the black hole is $x\sim\beta(H-\bar H)$ (the boundary energy fluctuation). Since $x$ ranges over all of $\R$ with no upper bound and the weight $\ee^x$ grows, $\tau(\id)=\int_{-\infty}^{\infty}\ee^x\dd x=\infty$, so the algebra is II$_\infty$, not II$_1$.\par
\textbf{\ref{ex:bh_crossed:2}.} In the black hole, $x=+\beta\,\delta E$ and the weight is $\ee^{x}=\ee^{\beta E}$ (more energy, more entropy). In de Sitter $x=-\beta q$ and the weight is $\ee^{-\beta q}$ (more observer energy, less entropy). The sign flip $x\to-x$ takes one to the other. In the black hole energy can be added without bound, so the total weight diverges; in de Sitter the observer's energy can only be added in the direction that lowers the entropy and is bounded below ($q\ge0$), so the total weight is finite.\par

\subsection*{\S\ref{ch:ew_qec} Entanglement wedges and error correction}
\textbf{\ref{ex:ew_qec:1}.} With JLMS, $K_B=\frac{\hat A}{4G}+K_{\rm bulk}$ on the code subspace, and FLM gives $S_B=\langle\frac{\hat A}{4G}\rangle+S_{\rm bulk}$. Then $S_{\rm rel}^{\rm bdy}=\Delta\langle K_B\rangle-\Delta S_B=\Delta\langle K_{\rm bulk}\rangle-\Delta S_{\rm bulk}=S_{\rm rel}^{\rm bulk}$: the area terms cancel.\par
\textbf{\ref{ex:ew_qec:2}.} A bulk operator in the entanglement wedge $\mathcal E_B$ is spacelike separated from, hence commutes with, every bulk operator in the complementary wedge $\mathcal E_{B'}$. On the code subspace, boundary operators in $B'$ are represented by bulk operators in $\mathcal E_{B'}$ (subregion duality), so they commute with bulk operators in $\mathcal E_B$.\par

\subsection*{\S\ref{ch:curved_hawking} Curved spacetime and Hawking radiation}
\textbf{\ref{ex:curved_hawking:1}.} $\kappa=1/4GM$ gives $T_H=\kappa/2\pi=1/(8\pi GM)$. The Hartle--Hawking state is KMS for Schwarzschild time at $\beta=2\pi/\kappa$, so the modular flow of the exterior algebra is the Schwarzschild time translation rescaled by $\beta$ (the Killing flow is the boost near the horizon, $\sigma_t=\mathrm{Ad}\,\ee^{-\ii\beta tH}$).\par
\textbf{\ref{ex:curved_hawking:2}.} Using $S_{\rm rel}(\lambda)=2\pi\int_\lambda^\infty(\lambda'-\lambda)T\,\dd\lambda'-\Delta S(\lambda)$, monotonicity gives $\partial_\lambda S_{\rm rel}=-2\pi\int_\lambda^\infty T\,\dd\lambda'-\partial_\lambda\Delta S\le0$. As $\lambda\to-\infty$ the local entropy variation $\partial_\lambda\Delta S\to0$ for a localized excitation, leaving $\int_{-\infty}^\infty T_{\lambda\lambda}\,\dd\lambda\ge0$: the ANEC, at the level of this formal argument.\par

\subsection*{\S\ref{ch:bhdS} Black holes and de Sitter: the sign of the thermodynamics}
\textbf{\ref{ex:bhdS:1}.} $r_s=2GM$, $\kappa=1/4GM$, $T_H=\kappa/2\pi=1/(8\pi GM)$. $A=4\pi r_s^2=16\pi G^2M^2$ so $S=A/4G=4\pi GM^2$ and $\dd S/\dd M=8\pi GM=1/T_H$. $C=\dd M/\dd T_H=(\dd T_H/\dd M)^{-1}=(-1/(8\pi GM^2))^{-1}=-8\pi GM^2$.\par
\textbf{\ref{ex:bhdS:2}.} The exponent is $\phi(M)=4\pi GM^2-\beta M$. $\phi'=8\pi GM-\beta=0$ at $M=\beta/8\pi G$, and $\phi''=8\pi G>0$, so the saddle is a minimum of $\phi$ along the real line: the integrand is smallest there, the signature of the negative heat capacity (thermodynamic instability of the canonical ensemble).\par
\textbf{\ref{ex:bhdS:3}.} $t=\int_M^0\frac{\dd M'}{\dd M'/\dd t}=\int_0^M15360\pi G^2M'^2\,\dd M'=5120\pi G^2M^3$. The temperature $T_H=1/(8\pi GM)$ diverges as $M\to0$ and reaches the Planck scale when $M\sim M_P$, where the semiclassical approximation (and the Stefan--Boltzmann rate) fails; so the final stage is not described consistently.\par
\textbf{\ref{ex:bhdS:4}.} $\int_0^\infty\dd E\,\ee^{S_{\rm dS}-\beta_{\rm dS}E-\beta'E}=\ee^{S_{\rm dS}}\int_0^\infty\ee^{-(\beta_{\rm dS}+\beta')E}\dd E=\frac{\ee^{S_{\rm dS}}}{\beta_{\rm dS}+\beta'}$, which converges iff $\beta_{\rm dS}+\beta'>0$, i.e.\ $\beta'>-\beta_{\rm dS}$. For every positive temperature $\beta'>0$ it converges.\par

\section*{Chapter~\ref{ch:ncg}: Intermezzo: Connes' noncommutative geometry}

\subsection*{\S\ref{ch:gelfand} Geometry from algebra}
\textbf{\ref{ex:gelfand:1}.} The kernel of a character $\chi$ is a maximal ideal $I$. If for every $x\in X$ some $f_x\in I$ had $f_x(x)\ne0$, then by compactness finitely many open sets $\{f_{x_i}\ne0\}$ cover $X$ and $\sum|f_{x_i}|^2=\sum f_{x_i}^*f_{x_i}\in I$ is strictly positive, hence invertible, contradicting $I\ne C(X)$. So there is $x$ with $f(x)=0$ for all $f\in I$; then $I\subset I_x$ and by maximality $I=I_x$ and $\chi(f)=f(x)$.\par
\textbf{\ref{ex:gelfand:2}.} With $U=\ee^{2\pi\ii x}$ and $V$ the unitary implementing the rotation by $2\pi\theta$ ($Vf(x)V^{-1}=f(x-\theta)$), $VUV^{-1}=\ee^{2\pi\ii(x-\theta)}=\ee^{-2\pi\ii\theta}U$, i.e.\ $UV=\ee^{2\pi\ii\theta}VU$; $U,V$ generate the crossed product, the noncommutative torus. This is the Weyl relation $U(a)V(b)=\ee^{\ii ab}V(b)U(a)$ with $ab=2\pi\theta$ fixed: for rational $\theta$ the algebra is a matrix bundle over a torus, for irrational $\theta$ it is simple.\par

\subsection*{\S\ref{ch:spectral_triples} Spectral triples}
\textbf{\ref{ex:spectral_triples:1}.} $[D,f]=-\ii f'$, so $\|[D,f]\|=\|f'\|_\infty\le1$ is the condition $|f'|\le1$. Then $d(p,q)=\sup\{|f(p)-f(q)|:|f'|\le1\}$. On a circle of length $L$ the extremal $f$ is the distance function from $p$ (suitably smoothed at the antipode), giving $d(p,q)=\min(|x_p-x_q|,L-|x_p-x_q|)$, the geodesic distance.\par
\textbf{\ref{ex:spectral_triples:2}.} $[D,f]_{12}=m(f_2-f_1)$ and $[D,f]_{21}=m(f_1-f_2)$, so $\|[D,f]\|=m|f_1-f_2|$. The condition $\le1$ gives $|f_1-f_2|\le1/m$, hence $d(1,2)=\sup|f_1-f_2|=1/m$.\par

\subsection*{\S\ref{ch:dixmier} Infinitesimals and the Dixmier trace}
\textbf{\ref{ex:dixmier:1}.} $\mu_n=1/n$ so $\sum_{n\le N}\mu_n=H_N=\log N+\gamma+o(1)\sim\log N$, which is the borderline for $\mathcal L^{1,\infty}$. The Dixmier trace is the limit of $\frac1{\log N}\sum_{n\le N}\mu_n$ along $\omega$, which is $1$.\par
\textbf{\ref{ex:dixmier:2}.} On the non-zero modes $|D|^{-1}$ has eigenvalues $1/|n|$, i.e.\ singular values $1,1,\frac12,\frac12,\dots$, so $\sum_{k\le N}\mu_k=2H_{N/2}\sim2\log N$ and $\Tr_\omega|D|^{-1}=2$. The formula $\Tr_\omega(f|D|^{-1})=c\int_{S^1}f\,\dd x/2\pi$ with $\int_{S^1}\dd x/2\pi=1$ gives $c=2$.\par

\subsection*{\S\ref{ch:spectral_action} The spectral action principle}
\textbf{\ref{ex:spectral_action:1}.} $D^2$ on $T^4=\R^4/\Lambda$ has eigenvalues $|k|^2$, $k\in\Lambda^*$, each with multiplicity $4$ (spinor components), so $\Tr\ee^{-tD^2}=4\sum_{k\in\Lambda^*}\ee^{-t|k|^2}$. Poisson resummation gives $4\frac{\mathrm{vol}}{(4\pi t)^2}\sum_{\gamma\in\Lambda}\ee^{-|\gamma|^2/4t}=\frac{4\,\mathrm{vol}}{16\pi^2t^2}+\mathcal O(\ee^{-c/t})$: only $a_0\propto\mathrm{vol}$ is non-zero, and all higher Seeley--DeWitt coefficients vanish.\par
\textbf{\ref{ex:spectral_action:2}.} With $f_k=\int_0^\infty x^{k-1}f(x)\,\dd x$ ($k>0$) and $f_0=f(0)$: for $f=\ee^{-x^2}$, $f_4=\int_0^\infty x^3\ee^{-x^2}\dd x=\frac12$, $f_2=\int_0^\infty x\ee^{-x^2}\dd x=\frac12$ and $f_0=f(0)=1$.\par

\subsection*{\S\ref{ch:almost_comm} Almost-commutative geometries and the Standard Model}
\textbf{\ref{ex:almost_comm:1}.} For $a=(a_1,a_2)$, $b=(b_1,b_2)$, $a[D_F,b]=\begin{pmatrix}0&ma_1(b_2-b_1)\\ma_2(b_1-b_2)&0\end{pmatrix}$, so sums of such terms give independent entries $\phi,\psi$; self-adjointness of $A$ forces $\psi=\bar\phi$ (for real $m$). Thus $D_F\to\begin{pmatrix}0&m+\phi\\\bar m+\bar\phi&0\end{pmatrix}$ with a single complex scalar. The eigenvalues are $\pm|m+\phi|$ and, for $f(x)=\ee^{-x^2}$, $\Tr f(D_F/\Lambda)=2\ee^{-|m+\phi|^2/\Lambda^2}=2-2\frac{|M|^2}{\Lambda^2}+\frac{|M|^4}{\Lambda^4}-\dots$ with $M=m+\phi$: a quartic potential $\big(\frac{|M|^2}{\Lambda^2}-1\big)^2+1$ with its minimum at $|M|=\Lambda$, a toy Higgs mexican hat.\par
\textbf{\ref{ex:almost_comm:2}.} $\mathfrak u(\C\oplus\mathbb H\oplus M_3(\C))=\mathfrak u(1)\oplus\mathfrak{su}(2)\oplus\mathfrak u(3)$, the Lie algebra of $U(1)\times SU(2)\times U(3)$. The unimodularity condition (trace of the generator in the fermion representation vanishes, equivalently the determinant of the action on $\HH_F$ is $1$) removes one $\mathfrak u(1)$: it forces the trace part of $\mathfrak u(3)$ to be tied to the $\mathfrak u(1)$ factor, so $\mathfrak u(3)\to\mathfrak{su}(3)$ plus a hypercharge $\mathfrak u(1)$ whose charges are fixed (fractional $\frac16,\frac23,-\frac13,-\frac12,-1$ for the quarks and leptons), which is exactly the anomaly-free hypercharge assignment.\par

\subsection*{\S\ref{ch:lorentzian_nc} Lorentzian signature and open problems}
\textbf{\ref{ex:lorentzian_nc:1}.} For a Lorentzian Dirac operator $D^2=-\partial_t^2+\Delta$ (up to sign conventions) is not bounded below, its spectrum extends to $-\infty$ as well as $+\infty$, so $\ee^{-tD^2}$ is unbounded and $\Tr\ee^{-tD^2}$ diverges; the heat kernel does not exist and the spectral action must be reformulated (e.g.\ by Wick rotation or a Krein-space framework).\par

\subsection*{\S\ref{ch:thermal_time} Thermal time (Connes--Rovelli)}
\textbf{\ref{ex:thermal_time:1}.} For the TFD, $\sigma_t=\mathrm{Ad}\,\ee^{-\ii\beta tH}$, so thermal (modular) time $t$ is related to physical time $s$ by $s=-\beta t$. For a Rindler observer with acceleration $a$ the modular parameter is rapidity ($\beta=2\pi$ per unit modular time), so proper time $\tau=\eta/a$ and $T_U=a/2\pi$. For Schwarzschild the Killing time at infinity has period $2\pi/\kappa$, $T_H=\kappa/2\pi$. Near the horizon a static observer has $a\simeq\kappa/\sqrt f$, and the Tolman relation $T_{\rm loc}=T_H/\sqrt f=a/2\pi=T_U$ connects them: both are the same modular flow measured in different proper times.\par
\textbf{\ref{ex:thermal_time:2}.} If $\omega$ is a trace, $\Delta=\id$ and $\sigma_t=\mathrm{id}$ for all $t$: nothing evolves in thermal time. A trace is the infinite-temperature state ($\beta=0$), so ``infinite temperature has no time'': time is generated by the departure of the state from a trace (the non-trivial $\Delta$).\par

\subsection*{\S\ref{ch:bridge} Speculative bridge}
\textbf{\ref{ex:bridge:1}.} $\A$ embeds in the ultrapower $R^\omega$ of the hyperfinite II$_1$ factor iff for every finite set $a_1,\dots,a_r\in\A$, every $k$ and every $\epsilon>0$ there are matrices $A_1,\dots,A_r\in M_n(\C)$ (some $n$) with $\|A_i\|\le\|a_i\|$ and $\big|\tau(a_{i_1}\cdots a_{i_\ell})-\frac1n\Tr(A_{i_1}\cdots A_{i_\ell})\big|<\epsilon$ for all words of length $\ell\le k$ (Connes embedding property for $\A$ with its trace $\tau$).\par

\subsection*{\S\ref{ch:ncgcomp} Two computations: the Dirac spectrum on the sphere and the fermion count}
\textbf{\ref{ex:ncgcomp:1}.} On the unit $S^n$ the Killing spinors $\nabla_X\psi=\pm\frac\ii2X\cdot\psi$ are Dirac eigenspinors with eigenvalue $\mp n/2$; for $n=2$ the eigenvalues are $\mp1$ and there are $2^{\lfloor n/2\rfloor}=2$ independent ones for each sign. This gives multiplicity $2$ at $m=1$, matching $2m$. In general the Camporesi--Higuchi spectrum on $S^2$ is $\pm m$, $m\ge1$, with multiplicity $2m$ for each sign (consistent with the Weyl law $N(\Lambda)\simeq2\Lambda^2=2\,\mathrm{vol}\Lambda^2/4\pi$).\par
\textbf{\ref{ex:ncgcomp:2}.} Euler--Maclaurin: $\sum_{m\ge1}f(m)=\int_0^\infty f-\frac{f(0)}2-\frac{f'(0)}{12}+\frac{f'''(0)}{720}-\dots$ with $f(m)=m\ee^{-tm^2}=m-tm^3+\dots$, so $f'(0)=1$, $f'''(0)=-6t$. Hence $\sum=\frac1{2t}-\frac1{12}-\frac t{120}+\mathcal O(t^2)$ and $\Tr\ee^{-tD^2}=4\sum=\frac2t-\frac13-\frac t{30}+\mathcal O(t^2)$. At $t=0.05$: $40-0.33333-0.00167=39.66500$, and the direct sum $\sum_{m\le2000}4m\ee^{-tm^2}$ is $39.66498$; the difference $2\times10^{-5}$ equals the next term $-4\cdot60t^2/30240$.\par
\textbf{\ref{ex:ncgcomp:3}.} $\mathbb H=\{q_0+q_1\mathbf i+q_2\mathbf j+q_3\mathbf k\}$; the map $q\mapsto\begin{pmatrix}a&b\\-\bar b&\bar a\end{pmatrix}$ with $a=q_0+\ii q_1$, $b=q_2+\ii q_3$ is a $^*$-isomorphism onto $\{\begin{pmatrix}a&b\\-\bar b&\bar a\end{pmatrix}\}\subset M_2(\C)$, and $qq^*=1$ iff $|a|^2+|b|^2=1$, so the unitary group is $SU(2)$. $U(M_3(\C))=U(3)$ is the definition. The unitary group of $\C\oplus\mathbb H\oplus M_3(\C)$ is $U(1)\times SU(2)\times U(3)$; unimodularity (determinant of the action on $\HH_F$ equal to $1$) imposes one relation among the $\mathfrak u(1)$ phases and leaves $U(1)_Y\times SU(2)\times SU(3)$, up to a finite quotient, the Standard Model gauge group.\par
